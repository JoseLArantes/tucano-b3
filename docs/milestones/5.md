## M5 Objectives

When M5 is complete:

1. Requests without an API key are limited to `default_requests_per_minute` (default: 100/min per IP).
2. Requests with a valid API key are limited to `api_key_requests_per_minute` (default: 1000/min per key).
3. Exceeding the limit returns HTTP 429 with a Portuguese error message and `Retry-After` header.
4. Rate limit state is tracked per client identity (IP or API key).
5. API keys are loaded from a static TOML file.
6. Health endpoint bypasses rate limiting.
7. Rate limiting can be disabled via config.

```rust
//! Token bucket rate limiter.
//!
//! Each client (identified by API key or IP) gets a bucket.
//! Tokens are replenished at a fixed rate.
//! Each request consumes one token.
//! When the bucket is empty, the request is rejected with 429.

use std::time::{Duration, Instant};

/// A single token bucket for one client.
#[derive(Debug, Clone)]
pub struct TokenBucket {
    /// Maximum number of tokens (burst capacity)
    max_tokens: f64,
    /// Current number of tokens
    tokens: f64,
    /// Token replenishment rate (tokens per second)
    refill_rate: f64,
    /// Last time tokens were replenished
    last_refill: Instant,
}

impl TokenBucket {
    /// Create a new token bucket.
    ///
    /// `requests_per_minute`: the sustained rate limit.
    /// Burst capacity equals the per-minute limit (allows short bursts).
    pub fn new(requests_per_minute: u32) -> Self {
        let max_tokens = requests_per_minute as f64;
        let refill_rate = requests_per_minute as f64 / 60.0; // tokens per second

        Self {
            max_tokens,
            tokens: max_tokens,
            refill_rate,
            last_refill: Instant::now(),
        }
    }

    /// Try to consume one token.
    ///
    /// Returns Ok(()) if a token was available.
    /// Returns Err(seconds_until_next_token) if the bucket is empty.
    pub fn try_acquire(&mut self) -> Result<(), f64> {
        self.refill();

        if self.tokens >= 1.0 {
            self.tokens -= 1.0;
            Ok(())
        } else {
            // Calculate seconds until next token
            let deficit = 1.0 - self.tokens;
            let wait_seconds = deficit / self.refill_rate;
            Err(wait_seconds)
        }
    }

    /// Replenish tokens based on elapsed time.
    fn refill(&mut self) {
        let now = Instant::now();
        let elapsed = now.duration_since(self.last_refill).as_secs_f64();
        let new_tokens = elapsed * self.refill_rate;

        self.tokens = (self.tokens + new_tokens).min(self.max_tokens);
        self.last_refill = now;
    }

    /// Get current token count (for metrics/debugging).
    pub fn available_tokens(&self) -> f64 {
        self.tokens
    }
}
```

```rust
//! API key management.
//!
//! Loads API keys from a static TOML file.
//! Each key has a name and an optional custom rate limit.

use serde::{Deserialize, Serialize};
use std::collections::HashMap;
use std::path::Path;
use std::sync::RwLock;
use thiserror::Error;

/// A single API key entry.
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct ApiKeyEntry {
    /// The API key string (sent in X-API-Key header)
    pub key: String,
    /// Human-readable name for logging
    pub name: String,
    /// Custom rate limit (requests per minute). Overrides default.
    #[serde(default)]
    pub rate_limit: Option<u32>,
}

/// API key file format.
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct ApiKeyFile {
    #[serde(default)]
    pub keys: Vec<ApiKeyEntry>,
}

/// Errors during API key loading.
#[derive(Error, Debug)]
pub enum ApiKeyError {
    #[error("API key file not found: {path}")]
    FileNotFound { path: String },

    #[error("Failed to read API key file: {0}")]
    Read(#[from] std::io::Error),

    #[error("Failed to parse API key file: {0}")]
    Parse(#[from] toml::de::Error),
}

/// Thread-safe API key registry.
pub struct ApiKeyRegistry {
    keys: RwLock<HashMap<String, ApiKeyEntry>>,
}

impl ApiKeyRegistry {
    /// Load API keys from a TOML file.
    ///
    /// If the file doesn't exist, returns an empty registry (no keys).
    pub fn load(path: &Path) -> Result<Self, ApiKeyError> {
        if !path.exists() {
            tracing::warn!(
                path = %path.display(),
                "Arquivo de API keys não encontrado, registry vazio"
            );
            return Ok(Self {
                keys: RwLock::new(HashMap::new()),
            });
        }

        let content = std::fs::read_to_string(path)?;
        let file: ApiKeyFile = toml::from_str(&content)?;

        let mut keys = HashMap::new();
        for entry in file.keys {
            keys.insert(entry.key.clone(), entry);
        }

        tracing::info!(
            count = keys.len(),
            "API keys carregadas"
        );

        Ok(Self {
            keys: RwLock::new(keys),
        })
    }

    /// Create an empty registry (no API keys).
    pub fn empty() -> Self {
        Self {
            keys: RwLock::new(HashMap::new()),
        }
    }

    /// Validate an API key and return its entry if valid.
    pub fn validate(&self, key: &str) -> Option<ApiKeyEntry> {
        let keys = self.keys.read().expect("RwLock poisoned");
        keys.get(key).cloned()
    }

    /// Check if a key exists.
    pub fn exists(&self, key: &str) -> bool {
        let keys = self.keys.read().expect("RwLock poisoned");
        keys.contains_key(key)
    }

    /// Get the rate limit for a key (custom or default).
    pub fn rate_limit_for(&self, key: &str, default: u32) -> u32 {
        let keys = self.keys.read().expect("RwLock poisoned");
        keys.get(key)
            .and_then(|entry| entry.rate_limit)
            .unwrap_or(default)
    }

    /// Get the total number of registered keys.
    pub fn count(&self) -> usize {
        let keys = self.keys.read().expect("RwLock poisoned");
        keys.len()
    }
}
```

```rust
//! Rate limit middleware layer.
//!
//! Tower middleware that enforces per-client rate limits.

use super::bucket::TokenBucket;
use super::keys::ApiKeyRegistry;
use crate::config::RateLimitConfig;
use axum::body::Body;
use axum::http::{Request, Response, StatusCode};
use dashmap::DashMap;
use serde_json::json;
use std::future::Future;
use std::net::IpAddr;
use std::pin::Pin;
use std::sync::Arc;
use std::task::{Context, Poll};
use tower::{Layer, Service};

/// Header for API key authentication.
pub const API_KEY_HEADER: &str = "X-API-Key";

/// Header indicating rate limit remaining.
pub const RATE_LIMIT_REMAINING: &str = "X-RateLimit-Remaining";

/// Header indicating rate limit total.
pub const RATE_LIMIT_LIMIT: &str = "X-RateLimit-Limit";

/// Header indicating seconds to wait before retry.
pub const RETRY_AFTER: &str = "Retry-After";

/// Shared rate limit state.
///
/// Manages per-client token buckets.
/// Thread-safe via DashMap.
pub struct RateLimitState {
    /// Per-client token buckets, keyed by client identity
    buckets: DashMap<String, TokenBucket>,
    /// API key registry
    api_keys: Arc<ApiKeyRegistry>,
    /// Configuration
    config: RateLimitConfig,
}

impl RateLimitState {
    pub fn new(api_keys: Arc<ApiKeyRegistry>, config: RateLimitConfig) -> Self {
        Self {
            buckets: DashMap::new(),
            api_keys,
            config,
        }
    }

    /// Identify the client from the request.
    ///
    /// Priority:
    /// 1. API key from X-API-Key header
    /// 2. Client IP address
    fn identify_client(&self, req: &Request<Body>) -> (String, u32) {
        // Check for API key
        if let Some(key_header) = req.headers().get(API_KEY_HEADER) {
            if let Ok(key) = key_header.to_str() {
                if let Some(entry) = self.api_keys.validate(key) {
                    let limit = entry.rate_limit
                        .unwrap_or(self.config.api_key_requests_per_minute);
                    return (format!("key:{}", key), limit);
                }
            }
        }

        // Fall back to IP
        let ip = req.extensions()
            .get::<axum::extract::ConnectInfo<std::net::SocketAddr>>()
            .map(|ci| ci.0.ip().to_string())
            .unwrap_or_else(|| "unknown".to_string());

        (format!("ip:{}", ip), self.config.default_requests_per_minute)
    }

    /// Try to acquire a token for the given client.
    ///
    /// Returns Ok(remaining_tokens) or Err(wait_seconds).
    pub fn try_acquire(&self, client_id: &str, limit: u32) -> Result<u32, f64> {
        let mut bucket = self.buckets
            .entry(client_id.to_string())
            .or_insert_with(|| TokenBucket::new(limit));

        match bucket.try_acquire() {
            Ok(()) => Ok(bucket.available_tokens() as u32),
            Err(wait) => Err(wait),
        }
    }

    /// Clean up buckets that haven't been used recently.
    ///
    /// Prevents unbounded memory growth from many unique IPs.
    pub fn cleanup(&self, max_buckets: usize) {
        if self.buckets.len() <= max_buckets {
            return;
        }

        // Remove buckets with very few tokens remaining (likely inactive)
        let to_remove: Vec<String> = self.buckets.iter()
            .filter(|entry| entry.value().available_tokens() < 1.0)
            .map(|entry| entry.key().clone())
            .collect();

        for key in to_remove {
            self.buckets.remove(&key);
        }
    }
}

/// Tower Layer for rate limiting.
#[derive(Clone)]
pub struct RateLimitLayer {
    state: Arc<RateLimitState>,
}

impl RateLimitLayer {
    pub fn new(state: Arc<RateLimitState>) -> Self {
        Self { state }
    }
}

impl<S> Layer<S> for RateLimitLayer {
    type Service = RateLimitMiddleware<S>;

    fn layer(&self, inner: S) -> Self::Service {
        RateLimitMiddleware {
            inner,
            state: Arc::clone(&self.state),
        }
    }
}

/// Tower Service that enforces rate limits.
#[derive(Clone)]
pub struct RateLimitMiddleware<S> {
    inner: S,
    state: Arc<RateLimitState>,
}

impl<S> Service<Request<Body>> for RateLimitMiddleware<S>
where
    S: Service<Request<Body>, Response = Response<Body>> + Clone + Send + 'static,
    S::Future: Send + 'static,
{
    type Response = S::Response;
    type Error = S::Error;
    type Future = Pin<Box<dyn Future<Output = Result<Self::Response, Self::Error>> + Send>>;

    fn poll_ready(&mut self, cx: &mut Context<'_>) -> Poll<Result<(), Self::Error>> {
        self.inner.poll_ready(cx)
    }

    fn call(&mut self, req: Request<Body>) -> Self::Future {
        let path = req.uri().path().to_string();

        // Skip rate limiting if disabled
        if !self.state.config.enabled {
            let mut inner = self.inner.clone();
            return Box::pin(async move { inner.call(req).await });
        }

        // Never rate-limit health endpoint
        if path == "/api/v1/health" || path == "/api/v1/metrics" {
            let mut inner = self.inner.clone();
            return Box::pin(async move { inner.call(req).await });
        }

        // Identify client
        let (client_id, limit) = self.state.identify_client(&req);

        // Try to acquire token
        match self.state.try_acquire(&client_id, limit) {
            Ok(remaining) => {
                let mut inner = self.inner.clone();
                Box::pin(async move {
                    let mut response = inner.call(req).await?;

                    // Add rate limit headers
                    response.headers_mut().insert(
                        RATE_LIMIT_REMAINING,
                        remaining.to_string().parse().expect("valid header"),
                    );
                    response.headers_mut().insert(
                        RATE_LIMIT_LIMIT,
                        limit.to_string().parse().expect("valid header"),
                    );

                    Ok(response)
                })
            }
            Err(wait_seconds) => {
                tracing::warn!(
                    client = %client_id,
                    path = %path,
                    wait = %wait_seconds,
                    "Limite de requisições excedido"
                );

                let retry_after = wait_seconds.ceil() as u64;
                let body = json!({
                    "error": {
                        "code": "RATE_LIMITED",
                        "message": "Limite de requisições excedido",
                        "details": format!("Aguarde {} segundo(s) e tente novamente.", retry_after),
                        "request_id": uuid::Uuid::new_v4().to_string()
                    }
                });

                let response = Response::builder()
                    .status(StatusCode::TOO_MANY_REQUESTS)
                    .header("content-type", "application/json")
                    .header(RETRY_AFTER, retry_after.to_string())
                    .body(Body::from(serde_json::to_string(&body).expect("valid json")))
                    .expect("valid response");

                Box::pin(async move { Ok(response) })
            }
        }
    }
}
```

```rust
pub fn build_router(
    config: &AppConfig,
    data_loader: Arc<crate::data::reload::DataLoader>,
    cache: Arc<crate::cache::store::CacheStore>,
    rate_limit_state: Arc<crate::middleware::rate_limit::layer::RateLimitState>,
) -> Router {
    let api_routes = crate::api::routes(data_loader, cache.clone());

    Router::new()
        .nest("/api/v1", api_routes)
        .layer(RateLimitLayer::new(rate_limit_state))  // ADD THIS (before cache)

        )))
```

```rust
fn identify_client(&self, req: &Request<Body>) -> (String, u32) {
    // Check for API key
    if let Some(key_header) = req.headers().get(API_KEY_HEADER) {
        if let Ok(key) = key_header.to_str() {
            let key = key.trim();
            if !key.is_empty() {
                if let Some(entry) = self.api_keys.validate(key) {
                    let limit = entry.rate_limit
                        .unwrap_or(self.config.api_key_requests_per_minute);
                    tracing::debug!(
                        key_name = %entry.name,
                        "Cliente identificado por API key"
                    );
                    return (format!("key:{}", key), limit);
                }
                // Invalid API key — fall through to IP-based limiting
                tracing::debug!("API key inválida, usando IP");
            }
        }
    }

    // Fall back to IP
    let ip = req.extensions()
        .get::<axum::extract::ConnectInfo<std::net::SocketAddr>>()
        .map(|ci| ci.0.ip().to_string())
        .unwrap_or_else(|| "unknown".to_string());

    (format!("ip:{}", ip), self.config.default_requests_per_minute)
}
```
