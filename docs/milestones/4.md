1. Repeated identical queries are served from memory in **< 1ms**.

5. Cache is purged on data reload.
6. Cache can be disabled via `config.toml` without breaking anything.
7. Health and metrics endpoints bypass cache.

```rust
//! Cache store implementation.
//!
//! Thread-safe, TTL-based, LRU-evicting cache.
//! Uses DashMap for O(1) concurrent reads.

use crate::config::CacheConfig;
use dashmap::DashMap;
use std::sync::atomic::{AtomicU64, Ordering};
use std::sync::Arc;
use std::time::{Duration, Instant};

/// A single cache entry with metadata.
#[derive(Debug, Clone)]
pub struct CacheEntry {
    /// Cached response body (serialized JSON bytes)
    pub body: Vec<u8>,
    /// HTTP status code
    pub status: u16,
    /// Content-Type header value
    pub content_type: String,
    /// When this entry was created
    pub created_at: Instant,
    /// When this entry was last accessed (for LRU)
    pub last_accessed: Instant,
    /// Time-to-live for this entry
    pub ttl: Duration,
}

impl CacheEntry {
    /// Check if this entry has expired.
    pub fn is_expired(&self) -> bool {
        self.created_at.elapsed() > self.ttl
    }
}

/// Thread-safe cache store with TTL and LRU eviction.
pub struct CacheStore {
    entries: DashMap<String, CacheEntry>,
    max_entries: usize,
    ttl: Duration,
    enabled: bool,
    hits: AtomicU64,
    misses: AtomicU64,
    evictions: AtomicU64,
}

impl CacheStore {
    /// Create a new CacheStore from configuration.
    pub fn new(config: &CacheConfig) -> Self {
        Self {
            entries: DashMap::new(),
            max_entries: config.max_entries,
            ttl: Duration::from_secs(config.ttl_seconds),
            enabled: config.enabled,
            hits: AtomicU64::new(0),
            misses: AtomicU64::new(0),
            evictions: AtomicU64::new(0),
        }
    }

    /// Create a disabled cache (no-op).
    pub fn disabled() -> Self {
        Self {
            entries: DashMap::new(),
            max_entries: 0,
            ttl: Duration::from_secs(0),
            enabled: false,
            hits: AtomicU64::new(0),
            misses: AtomicU64::new(0),
            evictions: AtomicU64::new(0),
        }
    }

    /// Get a cached entry by key.
    ///
    /// Returns None if:
    /// - Cache is disabled
    /// - Key not found
    /// - Entry has expired
    pub fn get(&self, key: &str) -> Option<CacheEntry> {
        if !self.enabled {
            return None;
        }

        match self.entries.get(key) {
            Some(entry) => {
                if entry.is_expired() {
                    // Remove expired entry
                    drop(entry);
                    self.entries.remove(key);
                    self.misses.fetch_add(1, Ordering::Relaxed);
                    None
                } else {
                    // Update last_accessed for LRU
                    // NOTE: DashMap doesn't allow mutation through get(),
                    // so we use entry API
                    drop(entry);
                    if let Some(mut entry) = self.entries.get_mut(key) {
                        entry.last_accessed = Instant::now();
                        self.hits.fetch_add(1, Ordering::Relaxed);
                        Some(entry.clone())
                    } else {
                        self.misses.fetch_add(1, Ordering::Relaxed);
                        None
                    }
                }
            }
            None => {
                self.misses.fetch_add(1, Ordering::Relaxed);
                None
            }
        }
    }

    /// Insert a new entry into the cache.
    ///
    /// If the cache is at capacity, evicts the least recently used entry.
    pub fn insert(&self, key: String, body: Vec<u8>, status: u16, content_type: String) {
        if !self.enabled {
            return;
        }

        // Evict if at capacity
        if self.entries.len() >= self.max_entries {
            self.evict_lru();
        }

        let entry = CacheEntry {
            body,
            status,
            content_type,
            created_at: Instant::now(),
            last_accessed: Instant::now(),
            ttl: self.ttl,
        };

        self.entries.insert(key, entry);
    }

    /// Evict the least recently used entry.
    fn evict_lru(&self) {
        let mut oldest: Option<(String, Instant)> = None;

        for entry in self.entries.iter() {
            let key = entry.key().clone();
            let accessed = entry.value().last_accessed;
            match &oldest {
                None => oldest = Some((key, accessed)),
                Some((_, oldest_time)) if accessed < *oldest_time => {
                    oldest = Some((key, accessed));
                }
                _ => {}
            }
        }

        if let Some((key, _)) = oldest {
            self.entries.remove(&key);
            self.evictions.fetch_add(1, Ordering::Relaxed);
            tracing::debug!(key = %key, "Cache: entrada evitada (LRU)");
        }
    }

    /// Purge all entries from the cache.
    ///
    /// Called on data reload to ensure stale data is not served.
    pub fn purge(&self) {
        let count = self.entries.len();
        self.entries.clear();
        tracing::info!(entries = count, "Cache: todas as entradas removidas");
    }

    /// Get current number of entries.
    pub fn len(&self) -> usize {
        self.entries.len()
    }

    /// Check if cache is empty.
    pub fn is_empty(&self) -> bool {
        self.entries.is_empty()
    }

    /// Check if cache is enabled.
    pub fn is_enabled(&self) -> bool {
        self.enabled
    }

    /// Get cache hit count.
    pub fn hit_count(&self) -> u64 {
        self.hits.load(Ordering::Relaxed)
    }

    /// Get cache miss count.
    pub fn miss_count(&self) -> u64 {
        self.misses.load(Ordering::Relaxed)
    }

    /// Get cache eviction count.
    pub fn eviction_count(&self) -> u64 {
        self.evictions.load(Ordering::Relaxed)
    }

    /// Get cache hit rate as a percentage (0.0 to 100.0).
    pub fn hit_rate(&self) -> f64 {
        let total = self.hits.load(Ordering::Relaxed) + self.misses.load(Ordering::Relaxed);
        if total == 0 {
            return 0.0;
        }
        (self.hits.load(Ordering::Relaxed) as f64 / total as f64) * 100.0
    }

    /// Remove expired entries.
    ///
    /// Called periodically or on demand to free memory.
    pub fn cleanup_expired(&self) {
        let expired_keys: Vec<String> = self.entries.iter()
            .filter(|entry| entry.value().is_expired())
            .map(|entry| entry.key().clone())
            .collect();

        for key in expired_keys {
            self.entries.remove(&key);
        }
    }
}
```

```rust
//! Cache key generation.
//!
//! Generates deterministic cache keys from HTTP request attributes.
//! Key = BLAKE3(method + path + sorted query params)

use blake3::Hasher;

/// Generate a cache key from HTTP request attributes.
///
/// The key is a BLAKE3 hash of:
/// - HTTP method (always GET for this API)
/// - Path (e.g., "/api/v1/cotahist/tickers/VALE3/quotes")
/// - Sorted query parameters (e.g., "dias=5&ordem=desc")
///
/// Sorting query params ensures "a=1&b=2" and "b=2&a=1" produce the same key.
pub fn generate_cache_key(method: &str, path: &str, query: &str) -> String {
    let mut hasher = Hasher::new();
    hasher.update(method.as_bytes());
    hasher.update(b"|");
    hasher.update(path.as_bytes());
    hasher.update(b"|");
    hasher.update(sort_query_params(query).as_bytes());
    hasher.finalize().to_hex().to_string()
}

/// Sort query parameters alphabetically for deterministic key generation.
///
/// "b=2&a=1" → "a=1&b=2"
fn sort_query_params(query: &str) -> String {
    if query.is_empty() {
        return String::new();
    }

    let mut params: Vec<&str> = query.split('&').collect();
    params.sort();
    params.join("&")
}

/// Determine if a path should be cached.
///
/// Health and metrics endpoints are never cached.
pub fn is_cacheable(path: &str) -> bool {
    !path.starts_with("/api/v1/health")
        && !path.starts_with("/api/v1/metrics")
        && !path.starts_with("/api/v1/info")
}
```

```rust
//! Cache middleware.
//!
//! Intercepts HTTP requests and serves cached responses when available.
//! Only caches GET requests to cacheable paths.

use crate::cache::key::{generate_cache_key, is_cacheable};
use crate::cache::store::CacheStore;
use axum::body::Body;
use axum::http::{Request, Response, StatusCode};
use std::future::Future;
use std::pin::Pin;
use std::sync::Arc;
use std::task::{Context, Poll};
use tower::{Layer, Service};

/// Tower Layer for cache middleware.
#[derive(Clone)]
pub struct CacheLayer {
    cache: Arc<CacheStore>,
}

impl CacheLayer {
    pub fn new(cache: Arc<CacheStore>) -> Self {
        Self { cache }
    }
}

impl<S> Layer<S> for CacheLayer {
    type Service = CacheMiddleware<S>;

    fn layer(&self, inner: S) -> Self::Service {
        CacheMiddleware {
            inner,
            cache: Arc::clone(&self.cache),
        }
    }
}

/// Tower Service that checks cache before forwarding to handler.
#[derive(Clone)]
pub struct CacheMiddleware<S> {
    inner: S,
    cache: Arc<CacheStore>,
}

impl<S> Service<Request<Body>> for CacheMiddleware<S>
where
    S: Service<Request<Body>, Response = Response<Body>> + Clone + Send + 'static,
    S::Future: Send + 'static,
{
    type Response = S::Response;
    type Error = S::Error;
    type Future = Pin<Box<dyn Future<Output = Result<Self::Response, Self::Error>> + Send>>;

    fn poll_ready(&mut self, cx: &mut Context<'_>) -> Poll<Result<(), Self::Error>> {
        self.inner.poll_ready(cx)
    }

    fn call(&mut self, req: Request<Body>) -> Self::Future {
        let method = req.method().to_string();
        let path = req.uri().path().to_string();
        let query = req.uri().query().unwrap_or("").to_string();

        // Only cache GET requests to cacheable paths
        if method != "GET" || !is_cacheable(&path) {
            let mut inner = self.inner.clone();
            return Box::pin(async move { inner.call(req).await });
        }

        let cache_key = generate_cache_key(&method, &path, &query);
        let cache = Arc::clone(&self.cache);

        // Check cache
        if let Some(entry) = cache.get(&cache_key) {
            tracing::debug!(key = %cache_key, path = %path, "Cache HIT");

            let response = Response::builder()
                .status(StatusCode::from_u16(entry.status).unwrap_or(StatusCode::OK))
                .header("content-type", &entry.content_type)
                .header("x-cache", "HIT")
                .body(Body::from(entry.body))
                .expect("valid response");

            return Box::pin(async move { Ok(response) });
        }

        tracing::debug!(key = %cache_key, path = %path, "Cache MISS");

        // Cache miss — forward to handler and cache the response
        let mut inner = self.inner.clone();

        Box::pin(async move {
            let response = inner.call(req).await?;

            // Only cache successful responses
            if response.status().is_success() {
                let status = response.status().as_u16();
                let content_type = response.headers()
                    .get("content-type")
                    .and_then(|v| v.to_str().ok())
                    .unwrap_or("application/json")
                    .to_string();

                // Read response body for caching
                let (parts, body) = response.into_parts();
                let body_bytes = hyper::body::to_bytes(body, usize::MAX)
                    .await
                    .unwrap_or_default();

                // Store in cache
                cache.insert(
                    cache_key,
                    body_bytes.to_vec(),
                    status,
                    content_type.clone(),
                );

                // Rebuild response with original body
                let response = Response::from_parts(parts)
                    .map(Body::from);

                // Add x-cache header
                let mut response = Response::builder()
                    .status(status)
                    .header("content-type", &content_type)
                    .header("x-cache", "MISS")
                    .body(Body::from(body_bytes))
                    .expect("valid response");

                return Ok(response);
            }

            Ok(response)
        })
    }
}
```
