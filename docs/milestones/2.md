```rust
//! API response envelope.
//!
//! Every successful response is wrapped in ApiResponse<T>.
//! Every error response is wrapped in ApiError (defined in src/error.rs).

use serde::Serialize;
use utoipa::ToSchema;

/// Standard response envelope for all successful API responses.
#[derive(Debug, Serialize, ToSchema)]
pub struct ApiResponse<T: Serialize> {
    pub data: T,
    pub meta: ResponseMeta,
}

/// Metadata included in every successful response.
#[derive(Debug, Serialize, ToSchema)]
pub struct ResponseMeta {
    /// Unique request identifier (UUID v4)
    pub request_id: String,
    /// ISO 8601 timestamp of the response
    pub timestamp: String,
    /// Whether the response was served from cache
    pub cached: bool,
    /// Query execution time in milliseconds
    pub query_time_ms: f64,
}

impl ResponseMeta {
    /// Create a new ResponseMeta for a non-cached response.
    pub fn new(query_time_ms: f64) -> Self {
        Self {
            request_id: uuid::Uuid::new_v4().to_string(),
            timestamp: chrono::Utc::now().to_rfc3339(),
            cached: false,
            query_time_ms,
        }
    }

    /// Create a new ResponseMeta for a cached response.
    pub fn cached(query_time_ms: f64) -> Self {
        Self {
            request_id: uuid::Uuid::new_v4().to_string(),
            timestamp: chrono::Utc::now().to_rfc3339(),
            cached: true,
            query_time_ms,
        }
    }
}

impl<T: Serialize> ApiResponse<T> {
    /// Wrap data in a standard response envelope.
    pub fn new(data: T, query_time_ms: f64) -> Self {
        Self {
            data,
            meta: ResponseMeta::new(query_time_ms),
        }
    }
}
```

```rust
//! Pagination model shared by all list endpoints.

use axum::extract::Query;
use serde::{Deserialize, Serialize};
use utoipa::{IntoParams, ToSchema};

/// Pagination query parameters.
///
/// Used by all list endpoints:
///   GET /api/v1/cotahist/tickers?pagina=1&por_pagina=50
///   GET /api/v1/cotahist/tickers/VALE3/quotes?pagina=1&por_pagina=252
#[derive(Debug, Deserialize, IntoParams)]
pub struct PaginationParams {
    /// Page number (1-indexed)
    #[serde(default = "default_page")]
    pub pagina: u32,

    /// Items per page
    #[serde(default = "default_per_page")]
    pub por_pagina: u32,
}

fn default_page() -> u32 { 1 }
fn default_per_page() -> u32 { 50 }

impl PaginationParams {
    /// Maximum allowed items per page.
    pub const MAX_PER_PAGE: u32 = 1000;

    /// Validate and clamp pagination parameters.
    pub fn validated(&self) -> Self {
        Self {
            pagina: self.pagina.max(1),
            por_pagina: self.por_pagina.clamp(1, Self::MAX_PER_PAGE),
        }
    }

    /// SQL OFFSET value.
    pub fn offset(&self) -> u32 {
        (self.validated().pagina - 1) * self.validated().por_pagina
    }

    /// SQL LIMIT value.
    pub fn limit(&self) -> u32 {
        self.validated().por_pagina
    }
}

/// Pagination metadata included in list responses.
#[derive(Debug, Serialize, ToSchema)]
pub struct PaginationMeta {
    pub total: usize,
    pub pagina: u32,
    pub por_pagina: u32,
    pub total_paginas: u32,
}

impl PaginationMeta {
    pub fn new(total: usize, pagina: u32, por_pagina: u32) -> Self {
        let total_paginas = if por_pagina > 0 {
            (total as f64 / por_pagina as f64).ceil() as u32
        } else {
            0
        };
        Self { total, pagina, por_pagina, total_paginas }
    }
}
```

```rust
//! COTAHIST service layer.
//!
//! Builds and executes SQL queries for all COTAHIST endpoints.
//! All methods are async and return typed results.

use crate::data::query_service::QueryService;
use crate::data::ticker_registry::TickerRegistry;
use crate::error::{AppError, AppResult};
use crate::models::cotahist::*;
use crate::models::pagination::PaginationParams;
use arrow::array::*;
use arrow::record_batch::RecordBatch;
use std::sync::Arc;

/// Service for COTAHIST data operations.
pub struct CotahistService;

impl CotahistService {
    /// List all tickers with pagination and optional search filter.
    pub fn list_tickers(
        registry: &TickerRegistry,
        pagina: u32,
        por_pagina: u32,
        busca: Option<&str>,
        ativo: bool,
    ) -> AppResult<(Vec<TickerResponse>, usize)> {
        let mut tickers: Vec<TickerResponse> = registry.list(ativo)
            .into_iter()
            .filter(|t| {
                if let Some(search) = busca {
                    t.ticker.to_uppercase().starts_with(&search.to_uppercase())
                } else {
                    true
                }
            })
            .map(|t| TickerResponse {
                ticker: t.ticker,
                isin: t.isin,
                company_name: t.company_name,
                first_seen: t.first_seen,
                last_seen: t.last_seen,
                is_active: t.is_active,
            })
            .collect();

        // Sort by ticker name for deterministic output
        tickers.sort_by(|a, b| a.ticker.cmp(&b.ticker));

        let total = tickers.len();
        let offset = ((pagina.max(1) - 1) * por_pagina) as usize;
        let limit = por_pagina as usize;

        let page: Vec<TickerResponse> = tickers.into_iter()
            .skip(offset)
            .take(limit)
            .collect();

        Ok((page, total))
    }

    /// Get detailed info for a single ticker.
    pub async fn get_ticker_detail(
        query_service: &QueryService,
        registry: &TickerRegistry,
        ticker: &str,
    ) -> AppResult<TickerDetailResponse> {
        let upper = ticker.to_uppercase();

        // Validate ticker exists
        let info = registry.get(&upper).ok_or_else(|| AppError::TickerNotFound {
            ticker: upper.clone(),
        })?;

        // Query latest price and trading days count
        let sql = format!(
            "SELECT COUNT(DISTINCT date) as trading_days, \
                    MAX(date) as latest_date, \
                    (SELECT pre_ult FROM cotahist WHERE ticker = '{}' ORDER BY date DESC LIMIT 1) as latest_price \
             FROM cotahist WHERE ticker = '{}'",
            upper, upper
        );

        let batches = query_service.execute_sql(&sql).await
            .map_err(|e| AppError::Internal(anyhow::anyhow!("{}", e)))?;

        let (trading_days, latest_date, latest_price) = Self::extract_ticker_detail(&batches);

        Ok(TickerDetailResponse {
            ticker: info.ticker,
            isin: info.isin,
            company_name: info.company_name,
            first_seen: info.first_seen,
            last_seen: info.last_seen,
            is_active: info.is_active,
            total_trading_days: trading_days,
            latest_price,
            latest_date,
        })
    }

    /// Get quote history for a ticker with filters.
    pub async fn get_quotes(
        query_service: &QueryService,
        registry: &TickerRegistry,
        ticker: &str,
        desde: Option<&str>,
        ate: Option<&str>,
        dias: Option<u32>,
        ordem: &str,
        pagination: &PaginationParams,
    ) -> AppResult<(Vec<QuoteResponse>, usize)> {
        let upper = ticker.to_uppercase();

        // Validate ticker
        if !registry.exists(&upper) {
            return Err(AppError::TickerNotFound { ticker: upper });
        }

        // Validate dates if provided
        if let Some(d) = desde {
            Self::validate_date(d)?;
        }
        if let Some(d) = ate {
            Self::validate_date(d)?;
        }
        if let (Some(d), Some(a)) = (desde, ate) {
            if d > a {
                return Err(AppError::DateRangeInvalid);
            }
        }

        // Build SQL
        let mut sql = format!(
            "SELECT ticker, date, pre_abr, pre_max, pre_min, pre_med, pre_ult, \
                    pre_ofe_compra, pre_ofe_venda, tot_neg, quat_tot, vol_tot, \
                    CASE WHEN LAG(pre_ult) OVER (PARTITION BY ticker ORDER BY date) IS NULL THEN NULL \
                         ELSE ROUND(((pre_ult - LAG(pre_ult) OVER (PARTITION BY ticker ORDER BY date)) \
                         / LAG(pre_ult) OVER (PARTITION BY ticker ORDER BY date)) * 100.0, 4) END as var_pct, \
                    CASE WHEN LAG(pre_ult) OVER (PARTITION BY ticker ORDER BY date) IS NULL THEN NULL \
                         ELSE ROUND(pre_ult - LAG(pre_ult) OVER (PARTITION BY ticker ORDER BY date), 4) END as var_abs \
             FROM cotahist WHERE ticker = '{}'",
            upper
        );

        // Date filters
        if let Some(d) = desde {
            sql.push_str(&format!(" AND date >= '{}'", d));
        }
        if let Some(a) = ate {
            sql.push_str(&format!(" AND date <= '{}'", a));
        }

        // Last N trading days
        if let Some(n) = dias {
            sql.push_str(&format!(
                " AND date IN (SELECT DISTINCT date FROM cotahist WHERE ticker = '{}' ORDER BY date DESC LIMIT {})",
                upper, n
            ));
        }

        // Count total (before pagination)
        let count_sql = format!("SELECT COUNT(*) FROM ({})", sql);
        let count_batches = query_service.execute_sql(&count_sql).await
            .map_err(|e| AppError::Internal(anyhow::anyhow!("{}", e)))?;
        let total = Self::extract_count(&count_batches);

        // Ordering and pagination
        let order = if ordem == "asc" { "ASC" } else { "DESC" };
        sql.push_str(&format!(" ORDER BY date {}", order));
        sql.push_str(&format!(" LIMIT {} OFFSET {}", pagination.limit(), pagination.offset()));

        let batches = query_service.execute_sql(&sql).await
            .map_err(|e| AppError::Internal(anyhow::anyhow!("{}", e)))?;

        let quotes = Self::extract_quotes(&batches);

        Ok((quotes, total))
    }

    /// Get the latest quote for a ticker.
    pub async fn get_latest_quote(
        query_service: &QueryService,
        registry: &TickerRegistry,
        ticker: &str,
    ) -> AppResult<LatestQuoteResponse> {
        let upper = ticker.to_uppercase();

        if !registry.exists(&upper) {
            return Err(AppError::TickerNotFound { ticker: upper });
        }

        let sql = format!(
            "SELECT ticker, date, pre_abr, pre_max, pre_min, pre_med, pre_ult, \
                    pre_ofe_compra, pre_ofe_venda, tot_neg, quat_tot, vol_tot \
             FROM cotahist WHERE ticker = '{}' ORDER BY date DESC LIMIT 2",
            upper
        );

        let batches = query_service.execute_sql(&sql).await
            .map_err(|e| AppError::Internal(anyhow::anyhow!("{}", e)))?;

        let quotes = Self::extract_quotes(&batches);

        if quotes.is_empty() {
            return Err(AppError::NoDataInRange);
        }

        let latest = &quotes[0];
        let var_pct = if quotes.len() > 1 {
            let prev = &quotes[1];
            match (latest.pre_ult, prev.pre_ult) {
                (Some(curr), Some(prev_val)) if prev_val != 0.0 => {
                    Some(((curr - prev_val) / prev_val * 100.0 * 10000.0).round() / 10000.0)
                }
                _ => None,
            }
        } else {
            None
        };

        let var_abs = if quotes.len() > 1 {
            match (latest.pre_ult, quotes[1].pre_ult) {
                (Some(curr), Some(prev_val)) => Some((curr - prev_val * 10000.0).round() / 10000.0),
                _ => None,
            }
        } else {
            None
        };

        Ok(LatestQuoteResponse {
            ticker: latest.ticker.clone(),
            data: latest.data.clone(),
            pre_abr: latest.pre_abr,
            pre_max: latest.pre_max,
            pre_min: latest.pre_min,
            pre_med: latest.pre_med,
            pre_ult: latest.pre_ult,
            pre_ofe_compra: latest.pre_ofe_compra,
            pre_ofe_venda: latest.pre_ofe_venda,
            tot_neg: latest.tot_neg,
            quat_tot: latest.quat_tot,
            vol_tot: latest.vol_tot,
            var_pct,
            var_abs,
        })
    }

    /// Get all quotes for a specific date.
    pub async fn get_quotes_by_date(
        query_service: &QueryService,
        date: &str,
        pagination: &PaginationParams,
    ) -> AppResult<(Vec<QuoteResponse>, usize)> {
        Self::validate_date(date)?;

        let count_sql = format!(
            "SELECT COUNT(*) FROM cotahist WHERE date = '{}'", date
        );
        let count_batches = query_service.execute_sql(&count_sql).await
            .map_err(|e| AppError::Internal(anyhow::anyhow!("{}", e)))?;
        let total = Self::extract_count(&count_batches);

        if total == 0 {
            return Err(AppError::NoDataInRange);
        }

        let sql = format!(
            "SELECT ticker, date, pre_abr, pre_max, pre_min, pre_med, pre_ult, \
                    pre_ofe_compra, pre_ofe_venda, tot_neg, quat_tot, vol_tot, \
                    NULL as var_pct, NULL as var_abs \
             FROM cotahist WHERE date = '{}' \
             ORDER BY ticker \
             LIMIT {} OFFSET {}",
            date, pagination.limit(), pagination.offset()
        );

        let batches = query_service.execute_sql(&sql).await
            .map_err(|e| AppError::Internal(anyhow::anyhow!("{}", e)))?;

        let quotes = Self::extract_quotes(&batches);

        Ok((quotes, total))
    }

    /// Validate a date string in YYYY-MM-DD format.
    fn validate_date(date: &str) -> AppResult<()> {
        chrono::NaiveDate::parse_from_str(date, "%Y-%m-%d")
            .map_err(|_| AppError::InvalidDate { value: date.to_string() })?;
        Ok(())
    }

    /// Extract QuoteResponse records from DataFusion RecordBatches.
    fn extract_quotes(batches: &[RecordBatch]) -> Vec<QuoteResponse> {
        let mut quotes = Vec::new();

        for batch in batches {
            for row in 0..batch.num_rows() {
                let ticker = Self::get_str(batch, row, "ticker").unwrap_or_default();
                let date_raw = Self::get_str(batch, row, "date").unwrap_or_default();
                // Convert date format if needed
                let data = if date_raw.len() == 10 { date_raw } else { date_raw };

                quotes.push(QuoteResponse {
                    ticker,
                    data,
                    pre_abr: Self::get_f64(batch, row, "pre_abr"),
                    pre_max: Self::get_f64(batch, row, "pre_max"),
                    pre_min: Self::get_f64(batch, row, "pre_min"),
                    pre_med: Self::get_f64(batch, row, "pre_med"),
                    pre_ult: Self::get_f64(batch, row, "pre_ult"),
                    pre_ofe_compra: Self::get_f64(batch, row, "pre_ofe_compra"),
                    pre_ofe_venda: Self::get_f64(batch, row, "pre_ofe_venda"),
                    tot_neg: Self::get_u32(batch, row, "tot_neg"),
                    quat_tot: Self::get_u64(batch, row, "quat_tot"),
                    vol_tot: Self::get_f64(batch, row, "vol_tot"),
                    var_pct: Self::get_f64(batch, row, "var_pct"),
                    var_abs: Self::get_f64(batch, row, "var_abs"),
                });
            }
        }

        quotes
    }

    /// Extract a count from a COUNT(*) query result.
    fn extract_count(batches: &[RecordBatch]) -> usize {
        if batches.is_empty() || batches[0].num_rows() == 0 {
            return 0;
        }
        let batch = &batches[0];
        if let Some(col) = batch.column(0).as_any().downcast_ref::<Int64Array>() {
            col.value(0) as usize
        } else {
            0
        }
    }

    /// Extract ticker detail from query result.
    fn extract_ticker_detail(batches: &[RecordBatch]) -> (Option<u32>, Option<String>, Option<f64>) {
        if batches.is_empty() || batches[0].num_rows() == 0 {
            return (None, None, None);
        }

        let idx = schema.index_of(col_name).ok()?;
        let col = batch.column(idx);
        if col.is_null(row) { return None; }
        col.as_any().downcast_ref::<StringArray>()
            .map(|arr| arr.value(row).to_string())
```
