```rust
//! API performance benchmarks.
//!
//! Measures throughput and latency for all endpoint types.
//! Run with: cargo bench

use criterion::{criterion_group, criterion_main, BenchmarkId, Criterion, Throughput};
use std::sync::Arc;

/// Benchmark: QueryService SQL execution (cold path).
fn bench_query_service_sql(c: &mut Criterion) {
    let fixture_dir = std::path::PathBuf::from("tests/fixtures");
    if !fixture_dir.exists() {
        eprintln!("Skipping: fixtures not found");
        return;
    }

    let config = tucano_b3::config::DataConfig {
        dir: fixture_dir,
        reload_interval_seconds: 0,
    };

    let cache = Arc::new(tucano_b3::cache::store::CacheStore::new(
        &tucano_b3::config::CacheConfig {
            enabled: false,
            ttl_seconds: 0,
            max_entries: 0,
        },
    ));

    let loader = match tucano_b3::data::reload::DataLoader::new(&config, cache) {
        Ok(l) => l,
        Err(e) => {
            eprintln!("Skipping: failed to load data: {}", e);
            return;
        }
    };

    let rt = tokio::runtime::Runtime::new().unwrap();

    let mut group = c.benchmark_group("query_service");
    group.throughput(Throughput::Elements(1));

    // Benchmark: simple ticker query
    group.bench_function("ticker_latest", |b| {
        b.iter(|| {
            rt.block_on(async {
                let service = loader.get_query_service().await;
                if service.has_table("cotahist") {
                    let _ = service.execute_sql(
                        "SELECT * FROM cotahist WHERE ticker = 'VALE3' ORDER BY date DESC LIMIT 1"
                    ).await;
                }
            });
        });
    });

    // Benchmark: aggregation query
    group.bench_function("ticker_aggregation", |b| {
        b.iter(|| {
            rt.block_on(async {
                let service = loader.get_query_service().await;
                if service.has_table("cotahist") {
                    let _ = service.execute_sql(
                        "SELECT ticker, AVG(pre_ult), SUM(vol_tot) FROM cotahist GROUP BY ticker"
                    ).await;
                }
            });
        });
    });

    // Benchmark: date range query
    group.bench_function("date_range_query", |b| {
        b.iter(|| {
            rt.block_on(async {
                let service = loader.get_query_service().await;
                if service.has_table("cotahist") {
                    let _ = service.execute_sql(
                        "SELECT * FROM cotahist WHERE date >= '2026-01-01' AND date <= '2026-01-31' ORDER BY date"
                    ).await;
                }
            });
        });
    });

    group.finish();
}

/// Benchmark: Cache operations.
fn bench_cache(c: &mut Criterion) {
    let config = tucano_b3::config::CacheConfig {
        enabled: true,
        ttl_seconds: 300,
        max_entries: 10_000,
    };
    let store = tucano_b3::cache::store::CacheStore::new(&config);

    // Pre-populate
    for i in 0..1000 {
        store.insert(
            format!("key_{}", i),
            b"cached_response_body".to_vec(),
            200,
            "application/json".to_string(),
        );
    }

    let mut group = c.benchmark_group("cache");

    group.bench_function("cache_hit", |b| {
        b.iter(|| store.get("key_500"));
    });

    group.bench_function("cache_miss", |b| {
        b.iter(|| store.get("nonexistent_key"));
    });

    group.bench_function("cache_insert", |b| {
        b.iter(|| {

                format!("key_{}", rand::random::<u32>()),
                b"new_data".to_vec(),
                200,

    });

    group.finish();
}
```

```rust
//! Query optimization utilities.
//!
//! Provides helpers for building efficient SQL queries
//! that leverage DataFusion's query planner.

/// Build an optimized query for latest ticker quote.
///
/// Uses LIMIT 2 to fetch only the rows needed for var_pct computation.
/// Avoids full table scans.
pub fn latest_quote_sql(ticker: &str) -> String {
    format!(
        "SELECT ticker, date, pre_abr, pre_max, pre_min, pre_med, pre_ult, \
                pre_ofe_compra, pre_ofe_venda, tot_neg, quat_tot, vol_tot \
         FROM cotahist \
         WHERE ticker = '{}' \
         ORDER BY date DESC \
         LIMIT 2",
        ticker.to_uppercase()
    )
}

/// Build an optimized query for quote history with date range.
///
/// Pushes date filter into Parquet reader via predicate pushdown.
/// Only selects columns needed by the API response.
pub fn quote_history_sql(
    ticker: &str,
    since: Option<&str>,
    until: Option<&str>,
    limit: u32,
    offset: u32,
    order: &str,
) -> String {
    let mut sql = format!(
        "SELECT ticker, date, pre_abr, pre_max, pre_min, pre_med, pre_ult, \
                pre_ofe_compra, pre_ofe_venda, tot_neg, quat_tot, vol_tot \
         FROM cotahist \
         WHERE ticker = '{}'",
        ticker.to_uppercase()
    );

    if let Some(s) = since {
        sql.push_str(&format!(" AND date >= '{}'", s));
    }
    if let Some(u) = until {
        sql.push_str(&format!(" AND date <= '{}'", u));
    }

    sql.push_str(&format!(" ORDER BY date {}", order));
    sql.push_str(&format!(" LIMIT {} OFFSET {}", limit, offset));

    sql
}

/// Build an optimized count query.
///
/// Uses COUNT(*) which DataFusion can optimize via parquet metadata
/// without reading actual row data.
pub fn count_sql(table: &str, where_clause: &str) -> String {
    if where_clause.is_empty() {
        format!("SELECT COUNT(*) FROM {}", table)
    } else {
        format!("SELECT COUNT(*) FROM {} WHERE {}", table, where_clause)
    }
}
```

## Performance Targets

## Architecture Decisions for Performance

### 1. Embedded DataFusion
DataFusion is embedded in the binary. No external database server.
Queries execute in-process with zero network overhead.

### 2. Parquet Columnar Storage
Data is stored in Parquet format with zstd compression.
Columnar layout enables predicate pushdown and column projection.

### 3. In-Memory Cache
Hot queries are cached in a DashMap with TTL and LRU eviction.
Cache hits serve responses in < 1ms.

### 4. Streaming Response
Large result sets are streamed, not buffered entirely in memory.

### 5. Connection-Free Design
No database connections. Parquet files are memory-mapped or streamed.
Zero connection pool overhead.

## Optimization Techniques Applied

1. **DataFusion configuration tuning**: batch_size=8192, target_partitions=4
2. **Predicate pushdown**: date filters pushed into Parquet reader
3. **Column projection**: only needed columns selected
4. **LIMIT optimization**: latest-quote queries fetch only 2 rows
5. **Cache body size limit**: responses > 1 MB not cached
6. **Path normalization**: prevents metric label explosion
7. **Atomic counters**: metrics use lock-free atomic operations
