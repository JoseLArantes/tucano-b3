**CREATE:** `.github/workflows/release.yml`

```yaml
name: Release

on:
  push:
    tags:
      - 'v*'

permissions:
  contents: write

env:
  CARGO_TERM_COLOR: always

jobs:
  test:
    name: Pre-release Tests
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: dtolnay/rust-toolchain@1.82.0
      - uses: Swatinem/rust-cache@v2
      - run: cargo test --all-features --all-targets
      - run: cargo clippy --all-targets --all-features -- -D warnings

  build:
    name: Build (${{ matrix.target }})
    needs: test
    strategy:
      fail-fast: false
      matrix:
        include:
          - target: x86_64-unknown-linux-gnu
            os: ubuntu-latest
            archive: tar.gz
          - target: aarch64-unknown-linux-gnu
            os: ubuntu-latest
            archive: tar.gz
          - target: x86_64-apple-darwin
            os: macos-latest
            archive: tar.gz
          - target: aarch64-apple-darwin
            os: macos-latest
            archive: tar.gz
          - target: x86_64-pc-windows-msvc
            os: windows-latest
            archive: zip
    runs-on: ${{ matrix.os }}
    steps:
      - uses: actions/checkout@v4

      - uses: dtolnay/rust-toolchain@1.82.0
        with:
          targets: ${{ matrix.target }}

      - uses: Swatinem/rust-cache@v2

      - name: Install cross-compilation tools (Linux ARM)
        if: matrix.target == 'aarch64-unknown-linux-gnu'
        run: |
          sudo apt-get update
          sudo apt-get install -y gcc-aarch64-linux-gnu
          echo "CARGO_TARGET_AARCH64_UNKNOWN_LINUX_GNU_LINKER=aarch64-linux-gnu-gcc" >> $GITHUB_ENV

      - name: Build release binary
        run: cargo build --release --target ${{ matrix.target }}

      - name: Package (Unix)
        if: matrix.archive == 'tar.gz'
        shell: bash
        run: |
          cd target/${{ matrix.target }}/release
          tar -czf ../../../tucano-b3-${{ github.ref_name }}-${{ matrix.target }}.tar.gz tucano-b3
          cd ../../..
          sha256sum tucano-b3-${{ github.ref_name }}-${{ matrix.target }}.tar.gz > tucano-b3-${{ github.ref_name }}-${{ matrix.target }}.tar.gz.sha256

      - name: Package (Windows)
        if: matrix.archive == 'zip'
        shell: bash
        run: |
          cd target/${{ matrix.target }}/release
          7z a ../../../tucano-b3-${{ github.ref_name }}-${{ matrix.target }}.zip tucano-b3.exe
          cd ../../..
          sha256sum tucano-b3-${{ github.ref_name }}-${{ matrix.target }}.zip > tucano-b3-${{ github.ref_name }}-${{ matrix.target }}.zip.sha256

      - name: Upload artifacts
        uses: actions/upload-artifact@v4
        with:
          name: tucano-b3-${{ matrix.target }}
          path: |
            tucano-b3-${{ github.ref_name }}-${{ matrix.target }}.*

  publish-github:
    name: GitHub Release
    needs: build
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4

      - name: Download all artifacts
        uses: actions/download-artifact@v4
        with:
          path: artifacts

      - name: Flatten artifacts
        run: |
          mkdir -p release-assets
          find artifacts -type f -exec mv {} release-assets/ \;
          ls -la release-assets/

      - name: Create GitHub Release
        uses: softprops/action-gh-release@v2
        with:
          files: release-assets/*
          generate_release_notes: true
          draft: false
          prerelease: false
        env:
          GITHUB_TOKEN: ${{ secrets.GITHUB_TOKEN }}

  publish-crates:
    name: Publish to crates.io
    needs: build
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: dtolnay/rust-toolchain@1.82.0
      - name: Verify publish (dry run)
        run: cargo publish --dry-run
      - name: Publish
        run: cargo publish
        env:
          CARGO_REGISTRY_TOKEN: ${{ secrets.CARGO_REGISTRY_TOKEN }}
```

# ============================================================

# Install build dependencies
RUN apt-get update && apt-get install -y \
    pkg-config \
    libssl-dev \
    && rm -rf /var/lib/apt/lists/*

WORKDIR /build

# Cache dependencies

RUN cargo build --release
RUN rm -rf src

# Build actual application
COPY src/ ./src/
RUN touch src/main.rs
RUN cargo build --release

# ============================================================
# Runtime stage
# ============================================================
FROM debian:bookworm-slim AS runtime

# Install minimal runtime dependencies
RUN apt-get update && apt-get install -y --no-install-recommends \
    ca-certificates \
    curl \
    && rm -rf /var/lib/apt/lists/* \
    && apt-get clean

# Create non-root user
RUN groupadd --gid 1000 tucano \
    && useradd --uid 1000 --gid 1000 --create-home --shell /bin/bash tucano

# Copy binary
COPY --from=builder /build/target/release/tucano-b3 /usr/local/bin/tucano-b3

# Copy default config
COPY config.toml /etc/tucano/config.toml

# Create data directory
RUN mkdir -p /data/processed /data/ledger \
    && chown -R tucano:tucano /data

USER tucano

EXPOSE 8080

# Health check
HEALTHCHECK --interval=30s --timeout=3s --start-period=5s --retries=3 \
    CMD curl -sf http://localhost:8080/api/v1/health || exit 1

# Labels

      org.opencontainers.image.description="API REST de alta performance para dados de mercado da B3" \
      org.opencontainers.image.licenses="MIT OR Apache-2.0" \

## M8 Risk Register

| Risk | Likelihood | Impact | Mitigation |
|---|---|---|---|
| Cross-compilation fails for aarch64-linux | Medium | Medium | Test in CI; use `cross` crate if needed |
| Docker image too large | Low | Medium | Multi-stage build; debian-slim base |
| `cargo publish` fails due to dependency issue | Low | High | Dry-run before actual publish |
| SHA256 mismatch in Homebrew formula | Medium | Medium | Automate SHA256 update in release script |
| Windows build fails due to platform-specific code | Low | Medium | CI tests on Windows; `#[cfg]` guards |
| Release binary missing on some platforms | Low | High | CI matrix with fail-fast: false |

## M8 Exit Gate

M8 is complete when:

> A developer can clone the repo, run `cargo build --release`, get a binary < 30 MB, start it, curl all endpoints successfully, build a Docker image < 100 MB, run it, verify the health check passes, push a git tag, and see a GitHub Release with 5 platform binaries and SHA256 checksums appear within minutes. The README.md is in Portuguese, README.dev.md is in English, and `cargo publish --dry-run` passes.

If true, the project is ready for 1.0 release.
