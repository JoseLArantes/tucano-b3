Update `build_router` to include Swagger UI:

```rust
pub fn build_router(
    config: &AppConfig,
    data_loader: Arc<crate::data::reload::DataLoader>,
    cache: Arc<crate::cache::store::CacheStore>,
    rate_limit_state: Arc<crate::middleware::rate_limit::layer::RateLimitState>,
) -> Router {
    let api_routes = crate::api::routes(data_loader, cache.clone());

    Router::new()
        .nest("/api/v1", api_routes)
        // OpenAPI spec endpoint
        .route("/api/docs/openapi.json", get(|| async {
            Json(ApiDoc::openapi())
        }))
        // Swagger UI
        .merge(
            SwaggerUi::new("/api/docs/")
                .url("/api/docs/openapi.json", ApiDoc::openapi())
        )
        .layer(RateLimitLayer::new(rate_limit_state))
        .layer(CacheLayer::new(cache))
        .layer(RequestIdLayer::new())
        .layer(CompressionLayer::new())
        .layer(CorsLayer::permissive())
        .layer(TimeoutLayer::new(Duration::from_secs(
            config.server.request_timeout_seconds,
        )))
}
```

Add `Json` import:

