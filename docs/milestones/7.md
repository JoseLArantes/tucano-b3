//! Prometheus metrics registry.
//!
//! All metrics are registered once at startup and shared across the application.
//! Thread-safe: uses atomic counters internally.

use prometheus::{
    Encoder, HistogramOpts, HistogramVec, IntCounterVec, IntGauge, Opts, Registry,
    TextEncoder,
};
use std::sync::Arc;

/// Application-wide Prometheus metrics.
///
/// Shared via Arc across all middleware and handlers.
/// All metrics use the `tucano_` prefix.
pub struct AppMetrics {
    pub registry: Registry,

    // === HTTP metrics ===
    /// Total HTTP requests, labeled by method, path, status.
    pub http_requests_total: IntCounterVec,

    /// HTTP request duration in seconds, labeled by method, path.
    pub http_request_duration_seconds: HistogramVec,

    /// HTTP requests currently in flight.
    pub http_requests_in_flight: IntGauge,

    // === Cache metrics ===
    /// Cache hits.
    pub cache_hits_total: IntCounterVec,

    /// Cache misses.
    pub cache_misses_total: IntCounterVec,

    /// Cache evictions.
    pub cache_evictions_total: IntCounterVec,

    /// Current cache entries.
    pub cache_entries: IntGauge,

    // === Rate limit metrics ===
    /// Requests rejected by rate limiter.
    pub rate_limited_total: IntCounterVec,

    // === Data metrics ===
    /// Number of Parquet files loaded.
    pub data_files_loaded: IntGauge,

    /// Total number of COTAHIST quotes.
    pub data_cotahist_quotes_total: IntGauge,

    /// Total number of IBOV data points.
    pub data_ibov_rows_total: IntGauge,

    /// Number of tickers in registry.
    pub data_tickers_total: IntGauge,

    // === Service metrics ===
    /// Service uptime in seconds.
    pub uptime_seconds: IntGauge,

    /// Service info (always 1, carries version label).
    pub service_info: IntGauge,
}

impl AppMetrics {
    /// Create and register all metrics.
    ///
    /// Called once at startup. Panics if metric registration fails
    /// (indicates a programming error, not a runtime condition).
    pub fn new() -> Self {
        let registry = Registry::new();

        // HTTP request counter
        let http_requests_total = IntCounterVec::new(
            Opts::new("tucano_http_requests_total", "Total HTTP requests")
                .namespace("tucano"),
            &["method", "path", "status"],
        ).expect("metric registration");
        registry.register(Box::new(http_requests_total.clone()))
            .expect("metric registration");

        // HTTP request duration histogram
        let http_request_duration_seconds = HistogramVec::new(
            HistogramOpts::new(
                "tucano_http_request_duration_seconds",
                "HTTP request duration in seconds",
            )
            .namespace("tucano")
            .buckets(vec![0.001, 0.005, 0.01, 0.025, 0.05, 0.1, 0.25, 0.5, 1.0, 2.5, 5.0]),
            &["method", "path"],
        ).expect("metric registration");
        registry.register(Box::new(http_request_duration_seconds.clone()))
            .expect("metric registration");

        // In-flight requests
        let http_requests_in_flight = IntGauge::new(
            "tucano_http_requests_in_flight",
            "HTTP requests currently being processed",
        ).expect("metric registration");
        registry.register(Box::new(http_requests_in_flight.clone()))
            .expect("metric registration");

        // Cache metrics
        let cache_hits_total = IntCounterVec::new(
            Opts::new("tucano_cache_hits_total", "Cache hits"),
            &[],
        ).expect("metric registration");
        registry.register(Box::new(cache_hits_total.clone()))
            .expect("metric registration");

        let cache_misses_total = IntCounterVec::new(
            Opts::new("tucano_cache_misses_total", "Cache misses"),
            &[],
        ).expect("metric registration");
        registry.register(Box::new(cache_misses_total.clone()))

        let cache_evictions_total = IntCounterVec::new(

            &[],
        ).expect("metric registration");
        registry.register(Box::new(cache_evictions_total.clone()))
            .expect("metric registration");

            "Current number of cache entries",

        registry.register(Box::new(rate_limited_total.clone()))
            .expect("metric registration");

        // Data metrics
        let data_files_loaded = IntGauge::new(
            "tucano_data_files_loaded",
            "Number of Parquet files loaded",
        ).expect("metric registration");
        registry.register(Box::new(data_files_loaded.clone()))
            .expect("metric registration");

        let data_cotahist_quotes_total = IntGauge::new(
            "tucano_data_cotahist_quotes_total",
            "Total COTAHIST quotes loaded",
        ).expect("metric registration");
        registry.register(Box::new(data_cotahist_quotes_total.clone()))
            .expect("metric registration");

        let data_ibov_rows_total = IntGauge::new(
            "tucano_data_ibov_rows_total",
            "Total IBOV data points loaded",

        registry.register(Box::new(data_ibov_rows_total.clone()))
            .expect("metric registration");

            "tucano_data_tickers_total",
            "Number of tickers in registry",
        ).expect("metric registration");
        registry.register(Box::new(data_tickers_total.clone()))
            .expect("metric registration");

        let uptime_seconds = IntGauge::new(

        ).expect("metric registration");

            .expect("metric registration");

            Opts::new("tucano_service_info", "Service information")
                .const_label("version", env!("CARGO_PKG_VERSION")),
        ).expect("metric registration");
        service_info.set(1);
        registry.register(Box::new(service_info.clone()))
            .expect("metric registration");

        Self {
            registry,
            http_requests_total,
            http_request_duration_seconds,
            http_requests_in_flight,
            cache_hits_total,
            cache_misses_total,
            cache_evictions_total,
            cache_entries,
            rate_limited_total,
            data_files_loaded,
            data_cotahist_quotes_total,
            data_ibov_rows_total,
            data_tickers_total,
            uptime_seconds,
            service_info,
        }
    }

    /// Render all metrics as Prometheus exposition format.
    pub fn render(&self) -> String {
        let encoder = TextEncoder::new();
        let metric_families = self.registry.gather();
        let mut buffer = Vec::new();
        encoder.encode(&metric_families, &mut buffer)
            .expect("encoding metrics");
        String::from_utf8(buffer).expect("valid UTF-8")
    }

    /// Update data metrics from the DataLoader.
    pub fn update_data_metrics(&self, files: usize, cotahist_rows: usize, ibov_rows: usize, tickers: usize) {
        self.data_files_loaded.set(files as i64);
        self.data_cotahist_quotes_total.set(cotahist_rows as i64);
        self.data_ibov_rows_total.set(ibov_rows as i64);
        self.data_tickers_total.set(tickers as i64);
    }

    /// Update cache metrics from the CacheStore.
    pub fn update_cache_metrics(&self, entries: usize, hits: u64, misses: u64, evictions: u64) {
        self.cache_entries.set(entries as i64);
        // Note: cache hits/misses are tracked via counters in middleware
    }

    /// Update uptime gauge.
    pub fn update_uptime(&self, seconds: u64) {
        self.uptime_seconds.set(seconds as i64);
    }
}

//! Prometheus metrics endpoint.
//!
//! GET /api/v1/metrics
//!
//! Returns metrics in Prometheus exposition format.
//! This endpoint is never cached and never rate-limited.

use axum::extract::State;
use axum::response::IntoResponse;
use axum::routing::get;
use axum::Router;
use std::sync::Arc;

use crate::metrics::AppMetrics;

/// Shared state for the metrics endpoint.
#[derive(Clone)]
pub struct MetricsState {
    pub metrics: Arc<AppMetrics>,
    pub start_time: std::time::Instant,
}

/// GET /api/v1/metrics
///
/// Retorna métricas no formato Prometheus.
/// Este endpoint nunca é cacheado e nunca sofre rate limiting.
async fn metrics_handler(
    State(state): State<MetricsState>,
) -> impl IntoResponse {
    // Update uptime before rendering
    let uptime = state.start_time.elapsed().as_secs();
    state.metrics.update_uptime(uptime);

    let body = state.metrics.render();

    (
        [(axum::http::header::CONTENT_TYPE, "text/plain; version=0.0.4; charset=utf-8")],
        body,
    )
}

/// Build metrics routes.
pub fn routes(metrics: Arc<AppMetrics>, start_time: std::time::Instant) -> Router {
    let state = MetricsState { metrics, start_time };

    Router::new()
        .route("/metrics", get(metrics_handler))
        .with_state(state)
}

//! Request metrics middleware.
//!
//! Records request count and duration for every HTTP request.
//! Labels: method, path (normalized), status code.

use crate::metrics::AppMetrics;
use axum::body::Body;
use axum::http::Request;
use std::future::Future;
use std::pin::Pin;
use std::sync::Arc;
use std::task::{Context, Poll};
use std::time::Instant;
use tower::{Layer, Service};

/// Tower Layer for request metrics.
#[derive(Clone)]
pub struct MetricsLayer {
    metrics: Arc<AppMetrics>,
}

impl MetricsLayer {
    pub fn new(metrics: Arc<AppMetrics>) -> Self {
        Self { metrics }
    }
}

impl<S> Layer<S> for MetricsLayer {
    type Service = MetricsMiddleware<S>;

    fn layer(&self, inner: S) -> Self::Service {
        MetricsMiddleware {
            inner,
            metrics: Arc::clone(&self.metrics),
        }
    }
}

/// Tower Service that records request metrics.
#[derive(Clone)]
pub struct MetricsMiddleware<S> {
    inner: S,
    metrics: Arc<AppMetrics>,
}

impl<S> Service<Request<Body>> for MetricsMiddleware<S>
where
    S: Service<Request<Body>, Response = axum::response::Response> + Clone + Send + 'static,
    S::Future: Send + 'static,
{
    type Response = S::Response;
    type Error = S::Error;
    type Future = Pin<Box<dyn Future<Output = Result<Self::Response, Self::Error>> + Send>>;

    fn poll_ready(&mut self, cx: &mut Context<'_>) -> Poll<Result<(), Self::Error>> {
        self.inner.poll_ready(cx)
    }

    fn call(&mut self, req: Request<Body>) -> Self::Future {
        let method = req.method().to_string();
        let path = normalize_path(req.uri().path());
        let start = Instant::now();

        self.metrics.http_requests_in_flight.inc();

        let mut inner = self.inner.clone();
        let metrics = Arc::clone(&self.metrics);

        Box::pin(async move {
            let response = inner.call(req).await?;

            let duration = start.elapsed().as_secs_f64();
            let status = response.status().as_u16().to_string();

            metrics.http_requests_in_flight.dec();
            metrics.http_requests_total
                .with_label_values(&[&method, &path, &status])
                .inc();
            metrics.http_request_duration_seconds
                .with_label_values(&[&method, &path])
                .observe(duration);

            Ok(response)
        })
    }
}

/// Normalize URL path for metric labels.
///
/// Replaces dynamic segments with placeholders to prevent label explosion.
/// Examples:
///   /api/v1/cotahist/tickers/VALE3 → /api/v1/cotahist/tickers/{ticker}
///   /api/v1/cotahist/dates/2026-01-01/quotes → /api/v1/cotahist/dates/{date}/quotes
fn normalize_path(path: &str) -> String {
    let segments: Vec<&str> = path.split('/').collect();
    let mut normalized = Vec::new();

    for (i, segment) in segments.iter().enumerate() {
        if segment.is_empty() {
            continue;
        }

        // Replace known dynamic segments
        let is_ticker = i > 0 && segments[i - 1] == "tickers" && !segment.is_empty();
        let is_date = i > 0 && segments[i - 1] == "dates" && !segment.is_empty();

        if is_ticker {
            normalized.push("{ticker}".to_string());
        } else if is_date {
            normalized.push("{date}".to_string());
        } else {
            normalized.push(segment.to_string());
        }
    }

    format!("/{}", normalized.join("/"))
}

```rust
//! Request logging middleware.
//!
//! Logs every request with structured context:
//! - request_id
//! - method
//! - path
//! - status code
//! - duration_ms
//! - cached (true/false)
//! - client_ip

use axum::body::Body;
use axum::http::Request;
use std::future::Future;
use std::pin::Pin;
use std::task::{Context, Poll};
use std::time::Instant;
use tower::{Layer, Service};

/// Tower Layer for request logging.
#[derive(Clone, Debug, Default)]
pub struct RequestLoggingLayer;

impl RequestLoggingLayer {
    pub fn new() -> Self {
        Self
    }
}

impl<S> Layer<S> for RequestLoggingLayer {
    type Service = RequestLoggingMiddleware<S>;

    fn layer(&self, inner: S) -> Self::Service {
        RequestLoggingMiddleware { inner }
    }
}

/// Tower Service that logs each request.
#[derive(Clone, Debug)]
pub struct RequestLoggingMiddleware<S> {
    inner: S,
}

impl<S> Service<Request<Body>> for RequestLoggingMiddleware<S>
where
    S: Service<Request<Body>, Response = axum::response::Response> + Clone + Send + 'static,
    S::Future: Send + 'static,
{
    type Response = S::Response;
    type Error = S::Error;
    type Future = Pin<Box<dyn Future<Output = Result<Self::Response, Self::Error>> + Send>>;

    fn poll_ready(&mut self, cx: &mut Context<'_>) -> Poll<Result<(), Self::Error>> {
        self.inner.poll_ready(cx)
    }

    fn call(&mut self, req: Request<Body>) -> Self::Future {
        let method = req.method().to_string();
        let path = req.uri().path().to_string();
        let query = req.uri().query().unwrap_or("").to_string();
        let request_id = req.headers()
            .get("x-request-id")
            .and_then(|v| v.to_str().ok())
            .unwrap_or("unknown")
            .to_string();
        let start = Instant::now();

        let mut inner = self.inner.clone();

        Box::pin(async move {
            let response = inner.call(req).await?;

            let duration_ms = start.elapsed().as_secs_f64() * 1000.0;
            let status = response.status().as_u16();
            let cached = response.headers()
                .get("x-cache")
                .and_then(|v| v.to_str().ok())
                .map(|v| v == "HIT")
                .unwrap_or(false);

            if status >= 500 {
                tracing::error!(
                    request_id = %request_id,
                    method = %method,
                    path = %path,
                    status = status,
                    duration_ms = %format!("{:.2}", duration_ms),
                    cached = cached,
                    "Requisição processada"
                );
            } else if status >= 400 {
                tracing::warn!(
                    request_id = %request_id,
                    method = %method,
                    path = %path,
                    status = status,
                    duration_ms = %format!("{:.2}", duration_ms),
                    cached = cached,
                    "Requisição processada"
                );
            } else {
                tracing::info!(
                    request_id = %request_id,
                    method = %method,
                    path = %path,
                    status = status,
                    duration_ms = %format!("{:.2}", duration_ms),
                    cached = cached,
                    "Requisição processada"
                );
            }

            Ok(response)
        })
    }
}
```
