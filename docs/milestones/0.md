```rust
//! Configuration module for tucano-b3.
//!
//! Configuration is loaded in the following priority order (highest first):
//! 1. Environment variables (TUCANO_*)
//! 2. Config file (config.toml)
//! 3. Built-in defaults
//!
//! All configuration values are immutable after startup.

use serde::{Deserialize, Serialize};
use std::path::{Path, PathBuf};
use thiserror::Error;

/// Top-level configuration for the tucano-b3 service.
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct AppConfig {
    pub server: ServerConfig,
    pub data: DataConfig,
    pub cache: CacheConfig,
    pub rate_limit: RateLimitConfig,
    pub logging: LoggingConfig,
    pub metrics: MetricsConfig,
}

/// HTTP server configuration.
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct ServerConfig {
    /// Bind address (e.g., "0.0.0.0")
    pub host: String,
    /// Port number (e.g., 8080)
    pub port: u16,
    /// Number of tokio worker threads (0 = auto-detect)
    pub workers: usize,
    /// Request timeout in seconds
    pub request_timeout_seconds: u64,
}

/// Data directory configuration.
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct DataConfig {
    /// Path to the data/ directory (contains processed/ and ledger/)
    pub dir: PathBuf,
    /// Auto-reload interval in seconds (0 = disabled)
    pub reload_interval_seconds: u64,
}

/// In-memory cache configuration.
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct CacheConfig {
    /// Whether caching is enabled
    pub enabled: bool,
    /// Time-to-live for cache entries in seconds
    pub ttl_seconds: u64,
    /// Maximum number of cache entries before LRU eviction
    pub max_entries: usize,
}

/// Rate limiting configuration.
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct RateLimitConfig {
    /// Whether rate limiting is enabled
    pub enabled: bool,
    /// Default requests per minute per IP
    pub default_requests_per_minute: u32,
    /// Requests per minute per API key
    pub api_key_requests_per_minute: u32,
}

/// Logging configuration.
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct LoggingConfig {
    /// Log level: trace, debug, info, warn, error
    pub level: String,
    /// Log format: json or pretty
    pub format: String,
}

/// Metrics configuration.
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct MetricsConfig {
    /// Whether Prometheus metrics are enabled
    pub enabled: bool,
}

/// Errors that can occur during configuration loading.
#[derive(Error, Debug)]
pub enum ConfigError {
    #[error("Failed to read config file: {0}")]
    FileRead(#[from] std::io::Error),

    #[error("Failed to parse config file: {0}")]
    Parse(#[from] toml::de::Error),

    #[error("Config file not found: {path}")]
    NotFound { path: String },

    #[error("Invalid configuration: {message}")]
    Invalid { message: String },
}

impl AppConfig {
    /// Load configuration from file and environment variables.
    ///
    /// Priority order (highest first):
    /// 1. Environment variables (TUCANO_SERVER_PORT, etc.)
    /// 2. Config file (config.toml)
    /// 3. Built-in defaults
    pub fn load(config_path: Option<&Path>) -> Result<Self, ConfigError> {
        let mut config = Self::default();

        // Layer 1: Load from file if it exists
        if let Some(path) = config_path {
            if path.exists() {
                let content = std::fs::read_to_string(path)?;
                let file_config: AppConfig = toml::from_str(&content)?;
                config = file_config;
            } else {
                return Err(ConfigError::NotFound {
                    path: path.display().to_string(),
                });
            }
        }

        // Layer 2: Override with environment variables
        config.apply_env_overrides();

        // Validate
        config.validate()?;

        Ok(config)
    }

    /// Apply environment variable overrides.
    ///
    /// Environment variables use the pattern: TUCANO_{SECTION}_{FIELD}
    /// Examples:
    ///   TUCANO_SERVER_PORT=9090
    ///   TUCANO_DATA_DIR=/mnt/data
    ///   TUCANO_CACHE_TTL=600
    ///   TUCANO_LOG_LEVEL=debug
    fn apply_env_overrides(&mut self) {
        if let Ok(val) = std::env::var("TUCANO_SERVER_HOST") {
            self.server.host = val;
        }
        if let Ok(val) = std::env::var("TUCANO_SERVER_PORT") {
            if let Ok(port) = val.parse() {
                self.server.port = port;
            }
        }
        if let Ok(val) = std::env::var("TUCANO_SERVER_WORKERS") {
            if let Ok(workers) = val.parse() {
                self.server.workers = workers;
            }
        }
        if let Ok(val) = std::env::var("TUCANO_DATA_DIR") {
            self.data.dir = PathBuf::from(val);
        }
        if let Ok(val) = std::env::var("TUCANO_CACHE_ENABLED") {
            self.cache.enabled = val == "true" || val == "1";
        }
        if let Ok(val) = std::env::var("TUCANO_CACHE_TTL") {
            if let Ok(ttl) = val.parse() {
                self.cache.ttl_seconds = ttl;
            }
        }
        if let Ok(val) = std::env::var("TUCANO_CACHE_MAX_ENTRIES") {
            if let Ok(max) = val.parse() {
                self.cache.max_entries = max;
            }
        }
        if let Ok(val) = std::env::var("TUCANO_LOG_LEVEL") {
            self.logging.level = val;
        }
        if let Ok(val) = std::env::var("TUCANO_LOG_FORMAT") {
            self.logging.format = val;
        }
    }

    /// Validate configuration values.
    fn validate(&self) -> Result<(), ConfigError> {
        if self.server.port == 0 {
            return Err(ConfigError::Invalid {
                message: "server.port cannot be 0".to_string(),
            });
        }
        if self.cache.ttl_seconds == 0 && self.cache.enabled {
            return Err(ConfigError::Invalid {
                message: "cache.ttl_seconds cannot be 0 when cache is enabled".to_string(),
            });
        }
        if self.cache.max_entries == 0 && self.cache.enabled {
            return Err(ConfigError::Invalid {
                message: "cache.max_entries cannot be 0 when cache is enabled".to_string(),
            });
        }
        Ok(())
    }
}

impl Default for AppConfig {
    fn default() -> Self {
        Self {
            server: ServerConfig {
                host: "0.0.0.0".to_string(),
                port: 8080,
                workers: 0, // auto-detect
                request_timeout_seconds: 30,
            },
            data: DataConfig {
                dir: PathBuf::from("./data"),
                reload_interval_seconds: 0,
            },
            cache: CacheConfig {
                enabled: true,
                ttl_seconds: 300,
                max_entries: 10_000,
            },
            rate_limit: RateLimitConfig {
                enabled: true,
                default_requests_per_minute: 100,
                api_key_requests_per_minute: 1000,
            },
            logging: LoggingConfig {
                level: "info".to_string(),
                format: "json".to_string(),
            },
            metrics: MetricsConfig {
                enabled: true,
            },
        }
    }
}
```

```rust
//! Error handling for tucano-b3.
//!
//! All user-facing error messages are in Brazilian Portuguese.
//! Error codes are stable English identifiers for programmatic handling.

use axum::http::StatusCode;
use axum::response::{IntoResponse, Response};
use axum::Json;
use serde::Serialize;
use thiserror::Error;

/// Top-level error type for the tucano-b3 service.
#[derive(Error, Debug)]
pub enum AppError {
    #[error("Ticker não encontrado: {ticker}")]
    TickerNotFound { ticker: String },

    #[error("Formato de data inválido: {value}")]
    InvalidDate { value: String },

    #[error("Intervalo de datas inválido")]
    DateRangeInvalid,

    #[error("Nenhum dado encontrado no período")]
    NoDataInRange,

    #[error("Limite de requisições excedido")]
    RateLimited,

    #[error("Erro interno do servidor")]
    Internal(#[from] anyhow::Error),

    #[error("Serviço indisponível")]
    ServiceUnavailable,

    #[error("Parâmetro inválido: {message}")]
    InvalidParam { message: String },
}

/// Error code for programmatic handling.
impl AppError {
    pub fn code(&self) -> &'static str {
        match self {
            Self::TickerNotFound { .. } => "TICKER_NOT_FOUND",
            Self::InvalidDate { .. } => "INVALID_DATE",
            Self::DateRangeInvalid => "DATE_RANGE_INVALID",
            Self::NoDataInRange => "NO_DATA_IN_RANGE",
            Self::RateLimited => "RATE_LIMITED",
            Self::Internal(_) => "INTERNAL_ERROR",
            Self::ServiceUnavailable => "SERVICE_UNAVAILABLE",
            Self::InvalidParam { .. } => "INVALID_PARAM",
        }
    }

    pub fn status_code(&self) -> StatusCode {
        match self {
            Self::TickerNotFound { .. } => StatusCode::NOT_FOUND,
            Self::InvalidDate { .. } => StatusCode::BAD_REQUEST,
            Self::DateRangeInvalid => StatusCode::BAD_REQUEST,
            Self::NoDataInRange => StatusCode::NOT_FOUND,
            Self::RateLimited => StatusCode::TOO_MANY_REQUESTS,
            Self::Internal(_) => StatusCode::INTERNAL_SERVER_ERROR,
            Self::ServiceUnavailable => StatusCode::SERVICE_UNAVAILABLE,
            Self::InvalidParam { .. } => StatusCode::BAD_REQUEST,
        }
    }

    pub fn details(&self) -> Option<&'static str> {
        match self {
            Self::TickerNotFound { .. } => Some("Verifique se o ticker existe nos dados ingeridos."),
            Self::InvalidDate { .. } => Some("Use o formato AAAA-MM-DD (ex: 2026-11-10)."),
            Self::DateRangeInvalid => Some("A data inicial não pode ser posterior à data final."),
            Self::NoDataInRange => Some("Verifique o intervalo de datas disponível nos dados."),
            Self::RateLimited => Some("Aguarde e tente novamente."),
            _ => None,
        }
    }
}

/// JSON error response body.
#[derive(Serialize)]
pub struct ErrorResponse {
    pub error: ErrorDetail,
}

#[derive(Serialize)]
pub struct ErrorDetail {
    pub code: String,
    pub message: String,
    #[serde(skip_serializing_if = "Option::is_none")]
    pub details: Option<String>,
    pub request_id: String,
}

/// Convert AppError into an HTTP response.
impl IntoResponse for AppError {
    fn into_response(self) -> Response {
        let status = self.status_code();
        let request_id = uuid::Uuid::new_v4().to_string();

        let body = ErrorResponse {
            error: ErrorDetail {
                code: self.code().to_string(),
                message: self.to_string(),
                details: self.details().map(|s| s.to_string()),
                request_id,
            },
        };

        (status, Json(body)).into_response()
    }
}

/// Type alias for Result with AppError.
pub type AppResult<T> = Result<T, AppError>;
```

```rust
//! HTTP server module for tucano-b3.
//!
//! Uses axum with tower middleware for routing, compression, and tracing.
//! Supports graceful shutdown via SIGTERM/SIGINT.

use crate::config::AppConfig;
use crate::error::AppError;
use axum::Router;
use std::net::SocketAddr;
use tokio::signal;
use tower_http::timeout::TimeoutLayer;
use tower_http::compression::CompressionLayer;
use tower_http::cors::CorsLayer;
use std::time::Duration;

/// Build the axum Router with all routes and middleware.
///
/// This function is called once at startup and returns a fully-configured router.
pub fn build_router(config: &AppConfig) -> Router {
    let api_routes = crate::api::routes();

    Router::new()
        .nest("/api/v1", api_routes)
        .layer(CompressionLayer::new())
        .layer(CorsLayer::permissive())
        .layer(TimeoutLayer::new(Duration::from_secs(
            config.server.request_timeout_seconds,
        )))
}

/// Start the HTTP server and block until shutdown signal.
///
/// Handles SIGTERM and SIGINT for graceful shutdown.
/// All in-flight requests are completed before the process exits.
pub async fn run(config: AppConfig) -> Result<(), AppError> {
    let router = build_router(&config);

    let addr: SocketAddr = format!("{}:{}", config.server.host, config.server.port)
        .parse()
        .map_err(|e| AppError::Internal(anyhow::anyhow!("Invalid bind address: {}", e)))?;

    tracing::info!(
        host = %config.server.host,
        port = config.server.port,
        "Iniciando servidor tucano-b3"
    );

    let listener = tokio::net::TcpListener::bind(addr)
        .await
        .map_err(|e| AppError::Internal(anyhow::anyhow!("Failed to bind: {}", e)))?;

    axum::serve(listener, router)
        .with_graceful_shutdown(shutdown_signal())
        .await
        .map_err(|e| AppError::Internal(anyhow::anyhow!("Server error: {}", e)))?;

    tracing::info!("Servidor tucano-b3 finalizado com sucesso");

    Ok(())
}

/// Wait for a shutdown signal (SIGTERM or SIGINT/Ctrl+C).
async fn shutdown_signal() {
    let ctrl_c = async {
        signal::ctrl_c()
            .await
            .expect("Failed to install Ctrl+C handler");
    };

    #[cfg(unix)]
    let terminate = async {
        signal::unix::signal(signal::unix::SignalKind::terminate())
            .expect("Failed to install SIGTERM handler")
            .recv()
            .await;
    };

    #[cfg(not(unix))]
    let terminate = std::future::pending::<()>();

    tokio::select! {
        _ = ctrl_c => {
            tracing::info!("Recebido SIGINT, iniciando shutdown gracioso");
        }
        _ = terminate => {
            tracing::info!("Recebido SIGTERM, iniciando shutdown gracioso");
        }
    }
}
```

```rust
//! Request ID middleware.
//!
//! Assigns a UUID v4 to every incoming request.
//! The ID is added to the response as `X-Request-Id` header.
//! The ID is also available to handlers via request extensions.

use axum::body::Body;
use axum::http::{Request, Response};
use std::future::Future;
use std::pin::Pin;
use std::task::{Context, Poll};
use tower::{Layer, Service};
use uuid::Uuid;

/// Header name for the request ID.
pub const REQUEST_ID_HEADER: &str = "X-Request-Id";

/// Tower Layer that adds request ID generation.
#[derive(Clone, Debug, Default)]
pub struct RequestIdLayer;

impl RequestIdLayer {
    pub fn new() -> Self {
        Self
    }
}

impl<S> Layer<S> for RequestIdLayer {
    type Service = RequestIdMiddleware<S>;

    fn layer(&self, inner: S) -> Self::Service {
        RequestIdMiddleware { inner }
    }
}

/// Tower Service that generates and propagates request IDs.
#[derive(Clone, Debug)]
pub struct RequestIdMiddleware<S> {
    inner: S,
}

impl<S> Service<Request<Body>> for RequestIdMiddleware<S>
where
    S: Service<Request<Body>, Response = Response<Body>> + Clone + Send + 'static,
    S::Future: Send + 'static,
{
    type Response = S::Response;
    type Error = S::Error;
    type Future = Pin<Box<dyn Future<Output = Result<Self::Response, Self::Error>> + Send>>;

    fn poll_ready(&mut self, cx: &mut Context<'_>) -> Poll<Result<(), Self::Error>> {
        self.inner.poll_ready(cx)
    }

    fn call(&mut self, mut req: Request<Body>) -> Self::Future {
        // Generate or extract request ID
        let request_id = req
            .headers()
            .get(REQUEST_ID_HEADER)
            .and_then(|v| v.to_str().ok())
            .map(|s| s.to_string())
            .unwrap_or_else(|| Uuid::new_v4().to_string());

        // Insert into request extensions for downstream handlers
        req.extensions_mut().insert(RequestId(request_id.clone()));

        let mut inner = self.inner.clone();

        Box::pin(async move {
            let mut response = inner.call(req).await?;

            // Add request ID to response headers
            response.headers_mut().insert(
                REQUEST_ID_HEADER,
                request_id.parse().expect("UUID is always valid header value"),
            );

            Ok(response)
        })
    }
}

/// Extension type for accessing request ID in handlers.
#[derive(Clone, Debug)]
pub struct RequestId(pub String);
```
