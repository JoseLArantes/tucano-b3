```rust
//! COTAHIST fixed-width parser.
//!
//! Parses B3 COTAHIST files (LATIN1, fixed-width, 249 bytes per line).
//! Supports all 3 known schema versions.
//!
//! Reference: B3 "Layout do Arquivo COTAHIST" documentation.

use chrono::NaiveDate;
use std::io::{BufRead, BufReader, Read};
use thiserror::Error;

/// COTAHIST record types.
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum RecordType {
    Header,  // TIPOREG = "00"
    Quote,   // TIPOREG = "01"
    Trailer, // TIPOREG = "02"
}

/// A parsed COTAHIST quote record.
#[derive(Debug, Clone)]
pub struct CotahistRecord {
    pub ticker: String,
    pub isin: Option<String>,
    pub company_name: Option<String>,
    pub date: NaiveDate,
    pub year: i32,
    pub market_type: Option<u8>,
    pub pre_abr: Option<f64>,
    pub pre_max: Option<f64>,
    pub pre_min: Option<f64>,
    pub pre_med: Option<f64>,
    pub pre_ult: Option<f64>,
    pub pre_ofe_compra: Option<f64>,
    pub pre_ofe_venda: Option<f64>,
    pub tot_neg: Option<u32>,
    pub quat_tot: Option<u64>,
    pub vol_tot: Option<f64>,
}

/// Errors during COTAHIST parsing.
#[derive(Error, Debug)]
pub enum CotahistParseError {
    #[error("IO error: {0}")]
    Io(#[from] std::io::Error),

    #[error("Invalid line length at line {line}: expected 249, got {actual}")]
    WrongLineLength { line: usize, actual: usize },

    #[error("Unknown record type at line {line}: {tipo}")]
    UnknownRecordType { line: usize, tipo: String },

    #[error("Invalid date at line {line}: {value}")]
    InvalidDate { line: usize, value: String },

    #[error("Invalid decimal at line {line}, field {field}: {value}")]
    InvalidDecimal { line: usize, field: String, value: String },

    #[error("Encoding error at line {line}: {message}")]
    EncodingError { line: usize, message: String },
}

/// Statistics from a parse operation.
#[derive(Debug, Clone, Default)]
pub struct ParseStats {
    pub lines_total: usize,
    pub lines_parsed: usize,
    pub lines_skipped: usize,
    pub lines_errored: usize,
    pub tickers_seen: std::collections::HashSet<String>,
    pub date_min: Option<NaiveDate>,
    pub date_max: Option<NaiveDate>,
}

/// COTAHIST parser.
///
/// Streaming: reads line-by-line, never loads entire file into memory.
pub struct CotahistParser;

impl CotahistParser {
    /// Parse a COTAHIST file from any Read source.
    ///
    /// Returns an iterator of parsed records and final statistics.
    /// Lines with errors are skipped (logged) rather than aborting.
    pub fn parse<R: Read>(
        reader: R,
    ) -> Result<(Vec<CotahistRecord>, ParseStats), CotahistParseError> {
        let buf_reader = BufReader::with_capacity(1_048_576, reader); // 1 MB buffer
        let mut records = Vec::new();
        let mut stats = ParseStats::default();

        for (line_num, line_result) in buf_reader.lines().enumerate() {
            let line = match line_result {
                Ok(l) => l,
                Err(e) => {
                    tracing::warn!(line = line_num + 1, error = %e, "Erro ao ler linha");
                    stats.lines_errored += 1;
                    continue;
                }
            };

            stats.lines_total += 1;

            // Skip empty lines
            if line.trim().is_empty() {
                stats.lines_skipped += 1;
                continue;
            }

            // Parse the line
            match Self::parse_line(&line, line_num + 1) {
                Ok(Some(record)) => {
                    // Update stats
                    stats.tickers_seen.insert(record.ticker.clone());
                    stats.date_min = Some(stats.date_min.map_or(record.date, |d| d.min(record.date)));
                    stats.date_max = Some(stats.date_max.map_or(record.date, |d| d.max(record.date)));
                    stats.lines_parsed += 1;
                    records.push(record);
                }
                Ok(None) => {
                    // Header or trailer — skip
                    stats.lines_skipped += 1;
                }
                Err(e) => {
                    tracing::warn!(line = line_num + 1, error = %e, "Erro ao parsear linha");
                    stats.lines_errored += 1;
                }
            }
        }

        Ok((records, stats))
    }

    /// Parse a single COTAHIST line.
    ///
    /// Returns:
    /// - Ok(Some(record)) for quote lines (TIPOREG = "01")
    /// - Ok(None) for header/trailer lines
    /// - Err for malformed lines
    fn parse_line(line: &str, line_num: usize) -> Result<Option<CotahistRecord>, CotahistParseError> {
        // Minimum length check
        if line.len() < 11 {
            return Err(CotahistParseError::WrongLineLength {
                line: line_num,
                actual: line.len(),
            });
        }

        // Record type (bytes 0-1)
        let tipo_reg = &line[0..2];
        let record_type = match tipo_reg {
            "00" => RecordType::Header,
            "01" => RecordType::Quote,
            "02" => RecordType::Trailer,
            _ => {
                return Err(CotahistParseError::UnknownRecordType {
                    line: line_num,
                    tipo: tipo_reg.to_string(),
                });
            }
        };

        // Only parse quote records
        if record_type != RecordType::Quote {
            return Ok(None);
        }

        // Full line length validation (249 bytes for V3 schema)
        if line.len() < 249 {
            return Err(CotahistParseError::WrongLineLength {
                line: line_num,
                actual: line.len(),
            });
        }

        // === Parse fields by position ===
        // Reference: B3 COTAHIST layout (V3, 2005+)

        // DATA_PREGAO (bytes 2-9, YYYYMMDD)
        let date_str = line[2..10].trim();
        let date = Self::parse_date(date_str, line_num)?;

        // CODBDI (bytes 10-12) — skip

        // CODNEG (bytes 12-24) — ticker
        let raw_ticker = line[12..24].trim();
        let ticker = Self::normalize_ticker(raw_ticker);

        // TPMERC (bytes 24-27) — market type
        let market_type = line[24..27].trim().parse::<u8>().ok();

        // NOMRES (bytes 27-39) — company name (LATIN1)
        let company_name = {
            let raw = &line[27..39];
            let name = raw.trim();
            if name.is_empty() { None } else { Some(name.to_string()) }
        };

        // ESPECI (bytes 39-49) — specification (ON/PN/UNT)

        // FAT_COTACAO (bytes 52-53) — decimal factor
        let fat_cotacao: u8 = line[52..53].trim().parse().unwrap_or(2);

        // PREABR (bytes 56-69) — opening price
        let pre_abr = Self::parse_implicit_decimal(&line[56..69], fat_cotacao, line_num, "PREABR")?;

        // PREMAX (bytes 69-82) — max price
        let pre_max = Self::parse_implicit_decimal(&line[69..82], fat_cotacao, line_num, "PREMAX")?;

        // PREMIN (bytes 82-95) — min price
        let pre_min = Self::parse_implicit_decimal(&line[82..95], fat_cotacao, line_num, "PREMIN")?;

        // PREMED (bytes 95-108) — average price
        let pre_med = Self::parse_implicit_decimal(&line[95..108], fat_cotacao, line_num, "PREMED")?;

        // PREULT (bytes 108-121) — last/closing price
        let pre_ult = Self::parse_implicit_decimal(&line[108..121], fat_cotacao, line_num, "PREULT")?;

        // PREOFC (bytes 121-134) — buy offer price
        let pre_ofe_compra = Self::parse_implicit_decimal(&line[121..134], fat_cotacao, line_num, "PREOFC")?;

        // PREOFV (bytes 134-147) — sell offer price
        let pre_ofe_venda = Self::parse_implicit_decimal(&line[134..147], fat_cotacao, line_num, "PREOFV")?;

        // TOTNEG (bytes 147-152) — trade count
        let tot_neg = line[147..152].trim().parse::<u32>().ok();

        // QUATOT (bytes 152-170) — volume (shares)
        let quat_tot = line[152..170].trim().parse::<u64>().ok();

        // VOLTOT (bytes 170-188) — volume (BRL)
        let vol_tot = Self::parse_implicit_decimal(&line[170..188], 0, line_num, "VOLTOT")?;

        // CODISIN (bytes 191-203) — ISIN code
        let isin = {
            let raw = if line.len() > 203 { &line[191..203] } else { "" };
            let code = raw.trim();
            if code.is_empty() { None } else { Some(code.to_string()) }
        };

        Ok(Some(CotahistRecord {
            ticker,
            isin,
            company_name,
            date,
            year: date.year(),
            market_type,
            pre_abr,
            pre_max,
            pre_min,
            pre_med,
            pre_ult,
            pre_ofe_compra,
            pre_ofe_venda,
            tot_neg,
            quat_tot,
            vol_tot,
        }))
    }

    /// Parse date from YYYYMMDD format.
    fn parse_date(s: &str, line_num: usize) -> Result<NaiveDate, CotahistParseError> {
        NaiveDate::parse_from_str(s, "%Y%m%d").map_err(|_| {
            CotahistParseError::InvalidDate {
                line: line_num,
                value: s.to_string(),
            }
        })
    }

    /// Parse implicit decimal field.
    ///
    /// COTAHIST stores prices as integers with implied decimal places.
    /// The decimal factor determines where the decimal point goes.
    /// Example: "0000000001234" with factor 2 → 12.34
    fn parse_implicit_decimal(
        s: &str,
        factor: u8,
        line_num: usize,
        field_name: &str,
    ) -> Result<Option<f64>, CotahistParseError> {
        let trimmed = s.trim();
        if trimmed.is_empty() || trimmed.chars().all(|c| c == '0') {
            return Ok(Some(0.0));
        }

        let raw: u64 = trimmed.parse().map_err(|_| {
            CotahistParseError::InvalidDecimal {
                line: line_num,
                field: field_name.to_string(),
                value: trimmed.to_string(),
            }
        })?;

        let value = raw as f64 / 10f64.powi(factor as i32);
        Ok(Some(value))
    }

    /// Normalize ticker by stripping market-type suffixes.
    ///
    /// Examples:
    /// - "PETR4" → "PETR4"
    /// - "PETR4F" → "PETR4" (futures reference)
    /// - "VALE3 " → "VALE3" (trailing space)
    fn normalize_ticker(raw: &str) -> String {
        let trimmed = raw.trim().to_uppercase();
        // Strip trailing 'F' (futures reference) or 'A' (old format)
        if trimmed.len() > 4 && (trimmed.ends_with('F') || trimmed.ends_with('A')) {
            trimmed[..trimmed.len() - 1].to_string()
        } else {
            trimmed
        }
    }
}

/// Detect COTAHIST file type from filename.
///
/// Returns: "yearly", "monthly", or "daily"
pub fn detect_file_type(filename: &str) -> &'static str {
    let upper = filename.to_uppercase();

    if upper.contains("COTAHIST_D") && upper.len() <= 25 {
        // COTAHIST_D2026.TXT → yearly
        "yearly"
    } else if upper.contains("COTAHIST_M") {
        // COTAHIST_M202601.TXT → monthly
        "monthly"
    } else if upper.contains("COTAHIST_") {
        // COTAHIST_20260102.TXT → daily
        "daily"
    } else {
        "unknown"
    }
}
```

```rust
//! Parquet writer for COTAHIST ingestion.
//!
//! Writes parsed COTAHIST records to Parquet files.
//! Output schema matches src/data/schema.rs::cotahist_schema().
//! Uses zstd compression and 1M row groups.

use crate::data::schema::cotahist_schema;
use crate::ingest::cotahist_parser::CotahistRecord;
use arrow::array::*;
use arrow::record_batch::RecordBatch;
use parquet::arrow::ArrowWriter;
use parquet::file::properties::{WriterProperties, WriterVersion};
use std::fs;
use std::path::Path;
use std::sync::Arc;
use thiserror::Error;

/// Errors during Parquet writing.
#[derive(Error, Debug)]
pub enum ParquetWriteError {
    #[error("IO error: {0}")]
    Io(#[from] std::io::Error),

    #[error("Arrow error: {0}")]
    Arrow(#[from] arrow::error::ArrowError),

    #[error("Parquet error: {0}")]
    Parquet(#[from] parquet::errors::ParquetError),

    #[error("Failed to create directory: {path}")]
    DirCreate { path: String },
}

/// Statistics from a write operation.
#[derive(Debug, Clone)]
pub struct WriteStats {
    pub rows_written: usize,
    pub file_path: String,
    pub file_size_bytes: u64,
}

/// Write COTAHIST records to a Parquet file.
///
/// The output file is partitioned by year:
///   data/processed/cotahist/cotahist-{YYYY}.parquet
///
/// If the file already exists, it is rewritten (idempotent).
pub fn write_cotahist_parquet(
    records: &[CotahistRecord],
    output_dir: &Path,
) -> Result<Vec<WriteStats>, ParquetWriteError> {
    // Group records by year
    let mut by_year: std::collections::HashMap<i32, Vec<&CotahistRecord>> =
        std::collections::HashMap::new();

    for record in records {
        by_year.entry(record.year).or_default().push(record);
    }

    let mut results = Vec::new();

    for (year, year_records) in &by_year {
        let file_path = output_dir.join(format!("cotahist-{}.parquet", year));

        // Create output directory if needed
        if let Some(parent) = file_path.parent() {
            fs::create_dir_all(parent).map_err(|_| ParquetWriteError::DirCreate {
                path: parent.display().to_string(),
            })?;
        }

        let stats = write_year_parquet(year_records, &file_path)?;
        results.push(stats);
    }

    Ok(results)
}

/// Write records for a single year to a Parquet file.

    );

    let date_arr = Date32Array::from(
        records.iter()
            .map(|r| r.date.num_days_from_ce())
            .collect::<Vec<_>>()
    );

    let year_arr = Int32Array::from(
        records.iter().map(|r| r.year).collect::<Vec<_>>()
    );

    let market_type_arr = UInt8Array::from(
        records.iter().map(|r| r.market_type).collect::<Vec<_>>()
    );

    let pre_abr_arr = Float64Array::from(
        records.iter().map(|r| r.pre_abr).collect::<Vec<_>>()
    );

    let pre_max_arr = Float64Array::from(
        records.iter().map(|r| r.pre_max).collect::<Vec<_>>()
    );

    let pre_min_arr = Float64Array::from(
        records.iter().map(|r| r.pre_min).collect::<Vec<_>>()
    );

    let pre_med_arr = Float64Array::from(
        records.iter().map(|r| r.pre_med).collect::<Vec<_>>()
    );

    let pre_ult_arr = Float64Array::from(
        records.iter().map(|r| r.pre_ult).collect::<Vec<_>>()
    );

    let pre_ofe_compra_arr = Float64Array::from(
        records.iter().map(|r| r.pre_ofe_compra).collect::<Vec<_>>()
    );

    let pre_ofe_venda_arr = Float64Array::from(
        records.iter().map(|r| r.pre_ofe_venda).collect::<Vec<_>>()
    );

    let tot_neg_arr = UInt32Array::from(
        records.iter().map(|r| r.tot_neg).collect::<Vec<_>>()
    );

    let quat_tot_arr = UInt64Array::from(
        records.iter().map(|r| r.quat_tot).collect::<Vec<_>>()
    );

    let vol_tot_arr = Float64Array::from(
        records.iter().map(|r| r.vol_tot).collect::<Vec<_>>()
    );

    // Build RecordBatch
```

```rust
//! ZIP file handler for ingestion.
//!
//! Decompresses uploaded ZIP files and extracts COTAHIST files.

use std::io::{Cursor, Read};
use std::path::{Path, PathBuf};
use thiserror::Error;
use zip::ZipArchive;

/// Errors during ZIP handling.
#[derive(Error, Debug)]
pub enum ZipError {
    #[error("IO error: {0}")]
    Io(#[from] std::io::Error),

    #[error("ZIP error: {0}")]
    Zip(#[from] zip::result::ZipError),

    #[error("No COTAHIST files found in ZIP")]
    NoCotahistFiles,

    #[error("File too large: {size} bytes (max: {max} bytes)")]
    FileTooLarge { size: usize, max: usize },
}

/// Maximum ZIP file size: 500 MB.
const MAX_ZIP_SIZE: usize = 500 * 1024 * 1024;

/// Extracted file from a ZIP archive.
pub struct ExtractedFile {
    pub filename: String,
    pub content: Vec<u8>,
}

/// Extract COTAHIST files from a ZIP archive.
///
/// Only extracts files matching COTAHIST patterns:
/// - COTAHIST_D*.TXT (yearly)
/// - COTAHIST_M*.TXT (monthly)
/// - COTAHIST_*.TXT (daily)
///
/// Returns extracted files ready for parsing.
pub fn extract_cotahist_from_zip(data: &[u8]) -> Result<Vec<ExtractedFile>, ZipError> {
    if data.len() > MAX_ZIP_SIZE {
        return Err(ZipError::FileTooLarge {
            size: data.len(),
            max: MAX_ZIP_SIZE,
        });
    }

    let cursor = Cursor::new(data);
    let mut archive = ZipArchive::new(cursor)?;

    let mut extracted = Vec::new();

    for i in 0..archive.len() {
        let mut file = archive.by_index(i)?;
        let filename = file.name().to_string();

        // Skip directories
        if filename.ends_with('/') {
            continue;
        }

        // Only process COTAHIST files
        let upper = filename.to_uppercase();
        if !upper.contains("COTAHIST") || !upper.ends_with(".TXT") {
            continue;
        }

        // Read file content
        let mut content = Vec::new();
        file.read_to_end(&mut content)?;

        extracted.push(ExtractedFile {
            filename: Path::new(&filename)
                .file_name()
                .map(|n| n.to_string_lossy().to_string())
                .unwrap_or(filename),
            content,
        });
    }

    if extracted.is_empty() {
        return Err(ZipError::NoCotahistFiles);
    }

    tracing::info!(
        count = extracted.len(),
        "Arquivos COTAHIST extraídos do ZIP"
    );

    Ok(extracted)
}

/// Scan a local folder for COTAHIST files.
///
/// Returns paths to all COTAHIST .TXT files found.
pub fn scan_folder_for_cotahist(folder: &Path) -> Result<Vec<PathBuf>, ZipError> {
    if !folder.exists() {
        return Err(ZipError::Io(std::io::Error::new(
            std::io::ErrorKind::NotFound,
            format!("Folder not found: {}", folder.display()),
        )));
    }

    let mut files = Vec::new();

    for entry in std::fs::read_dir(folder)? {
        let entry = entry?;
        let path = entry.path();

        if !path.is_file() {
            continue;
        }

        let filename = path.file_name()
            .map(|n| n.to_string_lossy().to_uppercase())
            .unwrap_or_default();

        if filename.contains("COTAHIST") && filename.ends_with(".TXT") {
            files.push(path);
        }
    }

    files.sort();

    if files.is_empty() {
        return Err(ZipError::NoCotahistFiles);
    }

    Ok(files)
}
```

```rust
//! Ingestion service.
//!
//! Orchestrates the full ingestion pipeline:
//! 1. Extract COTAHIST files (from ZIP or folder)
//! 2. Parse each file
//! 3. Filter out already-ingested dates (incremental)
//! 4. Write to Parquet
//! 5. Update SQLite ledger (tickers + ingested dates)
//! 6. Reload DataLoader

use crate::data::reload::DataLoader;
use crate::ingest::cotahist_parser::{CotahistParser, CotahistRecord, ParseStats};
use crate::ingest::parquet_writer;
use crate::ingest::zip_handler::{self, ExtractedFile};
use crate::ledger::Ledger;
use chrono::NaiveDate;
use std::collections::HashSet;
use std::path::{Path, PathBuf};
use std::sync::Arc;
use thiserror::Error;
use uuid::Uuid;

/// Errors during ingestion.
#[derive(Error, Debug)]
pub enum IngestError {
    #[error("ZIP error: {0}")]
    Zip(#[from] zip_handler::ZipError),

    #[error("Parse error: {0}")]
    Parse(#[from] crate::ingest::cotahist_parser::CotahistParseError),

    #[error("Parquet write error: {0}")]
    ParquetWrite(#[from] parquet_writer::ParquetWriteError),

    #[error("Ledger error: {0}")]
    Ledger(String),

    #[error("Data reload error: {0}")]
    Reload(String),

    #[error("IO error: {0}")]
    Io(#[from] std::io::Error),
}

/// Result of a single file ingestion.
#[derive(Debug, Clone)]
pub struct FileIngestResult {
    pub filename: String,
    pub records_parsed: usize,
    pub records_new: usize,
    pub records_skipped: usize,
    pub date_min: Option<NaiveDate>,
    pub date_max: Option<NaiveDate>,
    pub parquet_files: Vec<parquet_writer::WriteStats>,
}

/// Result of a full ingestion task.
#[derive(Debug, Clone)]
pub struct IngestResult {
    pub task_id: String,
    pub files_processed: usize,
    pub total_records_parsed: usize,
    pub total_records_new: usize,
    pub total_records_skipped: usize,
    pub file_results: Vec<FileIngestResult>,
    pub success: bool,
    pub error: Option<String>,
}

/// Ingestion service.
pub struct IngestService;

impl IngestService {
    /// Ingest from uploaded ZIP data.
    pub fn ingest_zip(
        data: &[u8],
        data_dir: &Path,
        data_loader: &Arc<DataLoader>,
    ) -> Result<IngestResult, IngestError> {
        let task_id = Uuid::new_v4().to_string();

        tracing::info!(task_id = %task_id, "Iniciando ingestão de ZIP");

        // Extract files from ZIP
        let extracted = zip_handler::extract_cotahist_from_zip(data)?;

        // Process each file
        Self::ingest_extracted_files(&task_id, &extracted, data_dir, data_loader)
    }

    /// Ingest from a local folder.
    pub fn ingest_folder(
        folder: &Path,
        data_dir: &Path,
        data_loader: &Arc<DataLoader>,
    ) -> Result<IngestResult, IngestError> {
        let task_id = Uuid::new_v4().to_string();

        tracing::info!(task_id = %task_id, folder = %folder.display(), "Iniciando ingestão de pasta");

        // Scan folder
        let files = zip_handler::scan_folder_for_cotahist(folder)?;

        // Read and process each file
        let extracted: Vec<ExtractedFile> = files.iter()
            .filter_map(|path| {
                let content = std::fs::read(path).ok()?;
                let filename = path.file_name()?
                    .to_string_lossy()
                    .to_string();
                Some(ExtractedFile { filename, content })
            })
            .collect();

        Self::ingest_extracted_files(&task_id, &extracted, data_dir, data_loader)
    }

    /// Process extracted COTAHIST files.
    fn ingest_extracted_files(
        task_id: &str,
        files: &[ExtractedFile],
        data_dir: &Path,
        data_loader: &Arc<DataLoader>,
    ) -> Result<IngestResult, IngestError> {
        let output_dir = data_dir.join("processed/cotahist");
        let ledger_path = data_dir.join("ledger/tucano.db");

        let mut result = IngestResult {
            task_id: task_id.to_string(),
            files_processed: 0,
            total_records_parsed: 0,
            total_records_new: 0,
            total_records_skipped: 0,
            file_results: Vec::new(),
            success: true,
            error: None,
        };

        // Load ledger for incremental check
        let ledger = Ledger::open(&ledger_path)
            .map_err(|e| IngestError::Ledger(e.to_string()))?;

        // Get already-ingested dates
        let ingested_dates = ledger.get_all_ingested_dates("cotahist")
            .map_err(|e| IngestError::Ledger(e.to_string()))?;
        let ingested_set: HashSet<NaiveDate> = ingested_dates.into_iter().collect();

        let mut all_new_records: Vec<CotahistRecord> = Vec::new();

        for file in files {
            tracing::info!(
                task_id = %task_id,
                file = %file.filename,
                "Processando arquivo"
            );

            // Parse the file
            let cursor = std::io::Cursor::new(&file.content);
            let (records, stats) = CotahistParser::parse(cursor)?;

            // Filter out already-ingested dates (incremental)
            let new_records: Vec<CotahistRecord> = records.into_iter()
                .filter(|r| !ingested_set.contains(&r.date))
                .collect();

            let records_parsed = stats.lines_parsed;
            let records_new = new_records.len();
            let records_skipped = records_parsed - records_new;

            tracing::info!(
                task_id = %task_id,
                file = %file.filename,
                parsed = records_parsed,
                new = records_new,
                skipped = records_skipped,
                "Arquivo processado"
            );

            let file_result = FileIngestResult {
                filename: file.filename.clone(),
                records_parsed,
                records_new,
                records_skipped,
                date_min: stats.date_min,
                date_max: stats.date_max,
                parquet_files: Vec::new(), // Will be filled after write
            };

            result.file_results.push(file_result);
            result.files_processed += 1;
            result.total_records_parsed += records_parsed;
            result.total_records_new += records_new;
            result.total_records_skipped += records_skipped;

            all_new_records.extend(new_records);
        }

        // Write new records to Parquet
        if !all_new_records.is_empty() {
            let write_stats = parquet_writer::write_cotahist_parquet(
                &all_new_records,
                &output_dir,
            )?;

            tracing::info!(
                task_id = %task_id,
                records = all_new_records.len(),
                files = write_stats.len(),
                "Parquet gravado"
            );

            // Update ledger with new ingested dates
            let new_dates: HashSet<NaiveDate> = all_new_records.iter()
                .map(|r| r.date)
                .collect();

            for date in &new_dates {
                ledger.insert_ingested_date("cotahist", "ingested", *date)
                    .map_err(|e| IngestError::Ledger(e.to_string()))?;
            }

            // Update ticker registry
            for record in &all_new_records {
                ledger.upsert_ticker(
                    &record.ticker,
                    record.isin.as_deref(),
                    record.company_name.as_deref(),
                    record.date,
                ).map_err(|e| IngestError::Ledger(e.to_string()))?;
            }

            // Reload DataLoader to pick up new data
            let rt = tokio::runtime::Handle::try_current();
            if let Ok(handle) = rt {
                let loader = Arc::clone(data_loader);
                handle.spawn(async move {
                    if let Err(e) = loader.reload().await {
                        tracing::error!(error = %e, "Erro ao recarregar dados");
                    }
                });
            }
        }

        tracing::info!(
            task_id = %task_id,
            files = result.files_processed,
            parsed = result.total_records_parsed,
            new = result.total_records_new,
            skipped = result.total_records_skipped,
            "Ingestão concluída"
        );

        Ok(result)
    }
}
```

```rust
///
/// Faz upload de um arquivo ZIP contendo arquivos COTAHIST para ingestão.
#[utoipa::path(
    post,
    path = "/ingest/upload",
    tag = "ingest",
    summary = "Upload de arquivo ZIP para ingestão",
    description = "Faz upload de um arquivo ZIP contendo arquivos COTAHIST (.TXT). \
                   O sistema descompacta, parseia e converte os dados para Parquet. \
                   A ingestão é incremental e idempotente.",
    request_body(
        content = String,
        content_type = "multipart/form-data",
    ),
    responses(
        (status = 200, description = "Ingestão concluída com sucesso", body = ApiResponse<IngestResponse>),
        (status = 400, description = "Arquivo inválido ou muito grande", body = crate::error::ErrorResponse),
        (status = 429, description = "Limite de requisições excedido", body = crate::error::ErrorResponse),
        (status = 500, description = "Erro interno do servidor", body = crate::error::ErrorResponse),
    ),
    security(
        ("ApiKeyAuth" = []),
    ),
)]
async fn upload_handler(
    State(state): State<Arc<AppState>>,
    mut multipart: Multipart,
) -> AppResult<Json<ApiResponse<IngestResponse>>> {
    let start = Instant::now();

    // Extract file from multipart
    let mut file_data: Option<Vec<u8>> = None;

    while let Some(field) = multipart.next_field().await.map_err(|e| {
        AppError::InvalidParam {
            message: format!("Erro ao ler upload: {}", e),
        }
    })? {
        if field.name() == Some("file") {
            let data = field.bytes().await.map_err(|e| {
                AppError::InvalidParam {
                    message: format!("Erro ao ler dados do arquivo: {}", e),
                }
            })?;
            file_data = Some(data.to_vec());
        }
    }

    let data = file_data.ok_or_else(|| AppError::InvalidParam {
        message: "Nenhum arquivo enviado. Use o campo 'file'.".to_string(),
    })?;

    // Validate file size (500 MB max)
    if data.len() > 500 * 1024 * 1024 {
        return Err(AppError::InvalidParam {
            message: "Arquivo muito grande (máximo 500 MB)".to_string(),
        });
    }

    // Run ingestion
    let data_dir = state.data_loader.data_config.dir.clone();
    let result = IngestService::ingest_zip(&data, &data_dir, &state.data_loader)
        .map_err(|e| AppError::Internal(anyhow::anyhow!("{}", e)))?;

    let elapsed = start.elapsed().as_secs_f64() * 1000.0;

    let message = if result.total_records_new > 0 {
        format!(
            "{} registros novos ingeridos ({} ignorados por já existirem)",
            result.total_records_new, result.total_records_skipped
        )
    } else {
        "Nenhum registro novo. Todos os dados já foram ingeridos anteriormente.".to_string()
    };

    Ok(Json(ApiResponse::new(
        IngestResponse {
            task_id: result.task_id,
            files_processed: result.files_processed,
            total_records_parsed: result.total_records_parsed,
            total_records_new: result.total_records_new,
            total_records_skipped: result.total_records_skipped,
            success: result.success,
            message,
        },
        elapsed,
    )))
}
```

```rust
impl Ledger {
    /// Get all ingested dates for a format.
    pub fn get_all_ingested_dates(&self, format: &str) -> Result<Vec<NaiveDate>, LedgerError> {
        let mut stmt = self.conn.prepare(
            "SELECT DISTINCT date FROM ingested_dates WHERE format = ?1"
        )?;

        let rows = stmt.query_map(rusqlite::params![format], |row| {
            let date_str: String = row.get(0)?;
            Ok(date_str)
        })?;

        let mut dates = Vec::new();
        for row in rows {
            let date_str = row?;
            if let Ok(date) = NaiveDate::parse_from_str(&date_str, "%Y-%m-%d") {
                dates.push(date);
            }
        }

        Ok(dates)
    }

    /// Insert an ingested date.
    pub fn insert_ingested_date(
        &self,
        format: &str,
        partition_key: &str,
        date: NaiveDate,
    ) -> Result<(), LedgerError> {
        self.conn.execute(
            "INSERT OR IGNORE INTO ingested_dates (format, partition_key, date, ingestion_id)
             VALUES (?1, ?2, ?3, NULL)",
            rusqlite::params![format, partition_key, date.format("%Y-%m-%d").to_string()],
        )?;
        Ok(())
    }

    /// Upsert a ticker in the registry.
    pub fn upsert_ticker(
        &self,
        ticker: &str,
        isin: Option<&str>,
        company_name: Option<&str>,
        date: NaiveDate,
    ) -> Result<(), LedgerError> {
        self.conn.execute(
            "INSERT INTO tickers (ticker, isin, company_name, first_seen, last_seen, is_active)
             VALUES (?1, ?2, ?3, ?4, ?4, 1)
             ON CONFLICT(ticker) DO UPDATE SET
                 last_seen = MAX(last_seen, ?4),
                 first_seen = MIN(first_seen, ?4),
                 isin = COALESCE(?2, isin),
                 company_name = COALESCE(?3, company_name)",
            rusqlite::params![ticker, isin, company_name, date.format("%Y-%m-%d").to_string()],
        )?;
        Ok(())
    }
}
```

```rust
//! End-to-end ingestion test.

use std::io::Write;
use tempfile::TempDir;
use zip::write::FileOptions;
use zip::ZipWriter;

/// Create a minimal COTAHIST ZIP for testing.
fn create_test_cotahist_zip() -> Vec<u8> {
    let mut zip_buffer = Vec::new();
    {
        let mut zip = ZipWriter::new(std::io::Cursor::new(&mut zip_buffer));
        let options = FileOptions::default();

        // Create a minimal COTAHIST file
        let header = format!("00COTAHIST.2026{}", " ".repeat(234));
        let quote = format!(
            "0120260102   VALE3       VALE S.A.    ON         2   \
             000000000620500000000062890000000006172\
             000000000623100000000062170000000006215\
             00000000062190024500000000029847362  \
             0000001847293440     BRVALEACNPA3 {}",
            " ".repeat(46)
        );
        let trailer = format!("02{}", " ".repeat(247));

        let content = format!("{}\n{}\n{}\n", header, quote, trailer);

        zip.start_file("COTAHIST_D2026.TXT", options).unwrap();
        zip.write_all(content.as_bytes()).unwrap();
        zip.finish().unwrap();
    }
    zip_buffer
}

#[tokio::test]
async fn test_e2e_zip_ingestion() {
    let temp = TempDir::new().unwrap();
    let data_dir = temp.path().to_path_buf();

    // Create data directory structure
    std::fs::create_dir_all(data_dir.join("processed/cotahist")).unwrap();
    std::fs::create_dir_all(data_dir.join("ledger")).unwrap();

    // Create test ZIP
    let zip_data = create_test_cotahist_zip();

    // Run ingestion directly (bypass HTTP for unit test)
    let cache = std::sync::Arc::new(
        tucano_b3::cache::store::CacheStore::new(&tucano_b3::config::CacheConfig {
            enabled: false,
            ttl_seconds: 0,
            max_entries: 0,
        })
    );

    let config = tucano_b3::config::DataConfig {
        dir: data_dir.clone(),
        reload_interval_seconds: 0,
    };

    let data_loader = std::sync::Arc::new(
        tucano_b3::data::reload::DataLoader::new(&config, cache).unwrap()
    );

    let result = tucano_b3::ingest::service::IngestService::ingest_zip(
        &zip_data,
        &data_dir,
        &data_loader,
    );

    assert!(result.is_ok());
    let result = result.unwrap();

    assert_eq!(result.files_processed, 1);
    assert!(result.total_records_parsed > 0);
    assert!(result.total_records_new > 0);
    assert!(result.success);

    // Verify Parquet file was created
    let parquet_path = data_dir.join("processed/cotahist/cotahist-2026.parquet");
    assert!(parquet_path.exists());

    // Verify idempotency: re-ingest same ZIP
    let result2 = tucano_b3::ingest::service::IngestService::ingest_zip(
        &zip_data,
        &data_dir,
        &data_loader,
    );

    assert!(result2.is_ok());
    let result2 = result2.unwrap();

    // Second ingestion should skip all records
    assert_eq!(result2.total_records_new, 0);
    assert_eq!(result2.total_records_skipped, result.total_records_parsed);
}
```
